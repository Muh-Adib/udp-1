   291→  const year = new Date().getFullYear()
   292→  const mkQuote = async (n: number, data: Record<string, unknown>, items: { description: string; qty: number; unitPrice: number }[]) => {
   293→    const subtotal = items.reduce((a, i) => a + i.qty * i.unitPrice, 0)
   294→    const discountPct = (data.discountPct as number) ?? 0
   295→    const discountAmount = Math.round(subtotal * discountPct / 100)
   296→    const taxPct = (data.taxPct as number) ?? 11
   297→    const taxAmount = Math.round((subtotal - discountAmount) * taxPct / 100)
   298→    return db.quotation.create({ data: {
   299→      ...(data as unknown as Prisma.QuotationUncheckedCreateInput),
   300→      code: `QUO-${year}-${String(n).padStart(4, '0')}`,
   301→      subtotal, discountAmount, taxAmount,
   302→      total: subtotal - discountAmount + taxAmount,
   303→      items: { create: items.map((it, idx) => ({ ...it, sortOrder: idx + 1 })) },
   304→    } })
