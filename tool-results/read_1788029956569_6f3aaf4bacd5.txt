   106→function InvoiceRow({ inv, onPay, onDetail }: { inv: InvoiceDTO; onPay: (inv: InvoiceDTO) => void; onDetail: (inv: InvoiceDTO) => void }) {
   107→  const overdue = isOverdue(inv)
   108→  const pct = inv.total > 0 ? Math.min(100, Math.max(0, (inv.paidAmount / inv.total) * 100)) : 0
   109→  const canPay = inv.status !== 'PAID' && inv.status !== 'CANCELLED'
   110→  return (
   111→    <div className="flex flex-col gap-3 p-4 transition-colors hover:bg-slate-50/70 md:flex-row md:items-center md:gap-5">
   112→      {/* Kiri: identitas */}
   113→      <div className="min-w-0 flex-1">
   114→        <div className="flex flex-wrap items-center gap-2">
   115→          <span className="font-mono text-[11px] text-slate-400">{inv.code}</span>
   116→          <BrandChip name={inv.brandName} color={inv.brandColor} size="xs" />
   117→          {overdue && <span className="rounded-full bg-rose-100 px-2 py-0.5 text-[10px] font-semibold text-rose-700">Lewat jatuh tempo</span>}
   118→        </div>
   119→        <p className="mt-0.5 truncate text-sm font-semibold text-slate-900">{inv.title}</p>
   120→        <div className="mt-0.5 flex flex-wrap items-center gap-x-3 gap-y-0.5 text-[11px] text-slate-500">
   121→          <span className="inline-flex items-center gap-1"><Building2 className="h-3 w-3" />{inv.companyName}</span>
   122→          {inv.dueDate && (
   123→            <span className={cn('inline-flex items-center gap-1', overdue ? 'font-semibold text-rose-600' : 'text-slate-500')}>
   124→              <CalendarClock className="h-3 w-3" />
   125→              Jatuh tempo {formatDate(inv.dueDate)}
   126→            </span>
   127→          )}
   128→        </div>
   129→      </div>
   130→
   131→      {/* Tengah: progres pembayaran */}
   132→      <div className="w-full shrink-0 md:w-60">
   133→        <div className="flex items-center justify-between gap-2 text-[11px]">
   134→          <span className="truncate text-slate-500">
   135→            Terbayar <span className="font-semibold text-slate-700">{formatMoney(inv.paidAmount, inv.currency, true)}</span>
   136→            {' '}/ {formatMoney(inv.total, inv.currency, true)}
   137→          </span>
   138→          <Badge variant="outline" className="h-4 shrink-0 px-1.5 text-[10px] font-medium text-slate-500">{inv.payments.length}× bayar</Badge>
   139→        </div>
   140→        <Progress value={pct} className="mt-1.5 h-1.5 bg-slate-100 [&>div]:bg-emerald-500" />
   141→      </div>
   142→
   143→      {/* Kanan: status + aksi */}
   144→      <div className="flex items-center gap-2 md:justify-end">
   145→        <StatusBadge status={inv.status} />
   146→        <Button
   147→          size="sm"
   148→          className="h-8 gap-1.5 bg-teal-600 text-white hover:bg-teal-700"
   149→          disabled={!canPay}
   150→          title={canPay ? 'Catat pembayaran untuk invoice ini' : 'Invoice lunas / dibatalkan'}
   151→          onClick={() => onPay(inv)}
   152→        >
   153→          <Wallet className="h-3.5 w-3.5" />
   154→          <span className="hidden sm:inline">Catat Pembayaran</span>
   155→          <span className="sm:hidden">Bayar</span>
   156→        </Button>
   157→        <Button
   158→          size="icon"
   159→          variant="ghost"
   160→          className="h-8 w-8 text-slate-400 hover:text-slate-700"
   161→          aria-label={`Lihat detail invoice ${inv.code}`}
   162→          title="Lihat detail"
   163→          onClick={() => onDetail(inv)}
   164→        >
   165→          <Eye className="h-4 w-4" />
   166→        </Button>
   167→      </div>
   168→    </div>
   169→  )
   170→}

   350→        description: e instanceof Error ? e.message : undefined,

   417→  const brandMax = summary ? Math.max(...summary.byBrand.map(b => b.invoiced), 1) : 1
   418→  const paySisa = payInv ? payInv.total - payInv.paidAmount : 0
