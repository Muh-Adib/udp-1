    67→/* ---------- Chip SLA + tingkat eskalasi (solid rose + Flame bila menunggu > 2× sla) ---------- */
    68→function SlaChip({ breach }: { breach: SlaBreach }) {
    69→  const escalated = breach.over > breach.sla
    70→  if (escalated) {
    71→    return (
    72→      <span
    73→        title={`Melewati ${breach.sla * 2} jam — eskalasi ke pemberi lead`}
    74→        className="inline-flex shrink-0 items-center gap-1 rounded-full bg-rose-600 px-1.5 py-0.5 text-[10px] font-semibold text-white"
    75→      >
    76→        <Flame className="h-3 w-3" /> ESKALASI +{breach.over}j
    77→      </span>
    78→    )
    79→  }
    80→  return (
    81→    <span
    82→      title={`Melewati SLA respons ${breach.brandName} (${breach.sla} jam)`}
    83→      className="shrink-0 rounded-full bg-rose-50 px-1.5 py-0.5 text-[10px] font-semibold text-rose-700"
    84→    >
    85→      SLA +{breach.over}j
    86→    </span>
    87→  )
