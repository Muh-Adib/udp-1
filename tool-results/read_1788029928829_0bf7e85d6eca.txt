    75→/** Kalkulasi total live (rumus sama dengan server: semua dibulatkan) */
    76→function computeTotals(items: EditorItem[], discountPct: string, taxPct: string) {
    77→  const subtotal = items.reduce((sum, it) => sum + Math.round(toNum(it.qty) * toNum(it.unitPrice)), 0)
    78→  const dp = Math.min(100, Math.max(0, toNum(discountPct)))
    79→  const tp = Math.max(0, toNum(taxPct))
    80→  const discountAmount = Math.round((subtotal * dp) / 100)
    81→  const taxAmount = Math.round(((subtotal - discountAmount) * tp) / 100)
    82→  return { subtotal, discountPct: dp, discountAmount, taxPct: tp, taxAmount, total: subtotal - discountAmount + taxAmount }
    83→}

    85→/** Mata uang yang paling banyak muncul (fallback IDR) — untuk KPI nilai gabungan */
    86→function dominantCurrency(list: QuotationDTO[]): string {
    87→  const counts = new Map<string, number>()
    88→  list.forEach((q) => counts.set(q.currency, (counts.get(q.currency) ?? 0) + 1))
    89→  let best = 'IDR'
    90→  let bestN = 0
    91→  counts.forEach((n, cur) => {
    92→    if (n > bestN) { best = cur; bestN = n }
    93→  })
    94→  return best
    95→}

   305→/** Seksi diskusi client di dialog detail penawaran — staff dapat membalas (identitas di-set server dari session) */
   306→function DiscussionSection({ quotationId }: { quotationId: string }) {
   307→  const { toast } = useToast()
   308→  const [comments, setComments] = useState<PortalCommentDTO[] | null>(null)
   309→  const [draft, setDraft] = useState('')
   310→  const [sending, setSending] = useState(false)
   311→  const listRef = React.useRef<HTMLDivElement>(null)
   312→
   313→  const loadComments = React.useCallback(async () => {
   314→    try {
   315→      const list = await portalApi.comments('QUOTATION', quotationId)
   316→      setComments(list)
   317→      requestAnimationFrame(() => {
   318→        const el = listRef.current
   319→        if (el) el.scrollTop = el.scrollHeight
   320→      })
   321→    } catch (e) {
   322→      toast({ variant: 'destructive', title: 'Gagal memuat diskusi', description: errMsg(e) })
   323→      setComments([])
   324→    }
   325→  }, [quotationId, toast])
   326→
   327→  React.useEffect(() => { void loadComments() }, [loadComments])
   328→
   329→  const send = async () => {
   330→    const body = draft.trim()
   331→    if (!body || sending || body.length > MAX_COMMENT_LEN) return
   332→    setSending(true)
   333→    try {
   334→      const created = await portalApi.addComment({ entityType: 'QUOTATION', entityId: quotationId, body })
   335→      setComments((prev) => [...(prev ?? []), created])
   336→      setDraft('')
   337→      toast({ title: 'Balasan terkirim' })
   338→    } catch (e) {
   339→      toast({ variant: 'destructive', title: 'Gagal mengirim balasan', description: errMsg(e) })
   340→    } finally {
   341→      setSending(false)
   342→    }
   343→  }
   344→
   345→  const canSend = draft.trim().length > 0 && draft.length <= MAX_COMMENT_LEN && !sending
   346→
   347→  return (
   348→    <section aria-label="Diskusi client" className="overflow-hidden rounded-xl border border-slate-200">
   349→      {/* Header */}
   350→      <div className="flex items-center justify-between gap-2 border-b border-slate-100 bg-white px-4 py-3">
   351→        <div className="flex min-w-0 items-center gap-2">
   352→          <span className="flex h-7 w-7 shrink-0 items-center justify-center rounded-lg bg-slate-900 text-white">
   353→            <MessageSquare className="h-3.5 w-3.5" />
   354→          </span>
   355→          <h3 className="text-sm font-semibold text-slate-900">Diskusi Client</h3>
   356→        </div>
   357→        {comments && comments.length > 0 && (
   358→          <span className="inline-flex shrink-0 items-center rounded-full bg-slate-100 px-2 py-0.5 text-[11px] font-semibold text-slate-600">
   359→            {comments.length} pesan
   360→          </span>
   361→        )}
   362→      </div>
   363→
   364→      {/* Daftar pesan */}
   365→      <div ref={listRef} className={cn('max-h-64 space-y-3 overflow-y-auto bg-slate-50/60 px-4 py-3', SCROLLBAR)}>
   366→        {comments === null ? (
   367→          <div className="space-y-3 py-1">
   368→            <Skeleton className="h-12 w-3/4 rounded-2xl" />
   369→            <Skeleton className="ml-auto h-10 w-1/2 rounded-2xl" />
   370→            <Skeleton className="h-12 w-2/3 rounded-2xl" />
   371→          </div>
   372→        ) : comments.length === 0 ? (
   373→          <div className="py-6 text-center">
   374→            <MessageSquare className="mx-auto h-5 w-5 text-slate-300" />
   375→            <p className="mt-1.5 text-xs font-medium text-slate-500">Belum ada pesan dari client</p>
   376→            <p className="mt-0.5 text-[11px] text-slate-400">Pertanyaan client tentang penawaran ini akan tampil di sini.</p>
   377→          </div>
   378→        ) : (
   379→          comments.map((c) => <CommentBubble key={c.id} c={c} />)
   380→        )}
   381→      </div>
   382→
   383→      {/* Input balasan */}
   384→      <div className="border-t border-slate-100 bg-white px-4 py-3">
   385→        <div className="flex items-end gap-2">
   386→          <Textarea
   387→            value={draft}
   388→            onChange={(e) => setDraft(e.target.value)}
   389→            onKeyDown={(e) => {
   390→              if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') {
   391→                e.preventDefault()
   392→                if (canSend) void send()
   393→              }
   394→            }}
   395→            placeholder="Tulis balasan untuk client…"
   396→            aria-label="Tulis balasan"
   397→            rows={1}
   398→            className="max-h-24 min-h-[2.5rem] resize-none"
   399→          />
   400→          <Button
   401→            onClick={() => void send()}
   402→            disabled={!canSend}
   403→            className="h-10 shrink-0 gap-1.5 bg-slate-900 text-white hover:bg-slate-800"
   404→          >
   405→            {sending ? <Loader2 className="h-4 w-4 animate-spin" /> : <Send className="h-4 w-4" />}
   406→            Kirim
   407→          </Button>
   408→        </div>
   409→        {draft.length > 1800 && (
   410→          <p className={cn('mt-1 text-right text-[10px]', draft.length > MAX_COMMENT_LEN ? 'font-semibold text-rose-600' : 'text-slate-400')}>
   411→            {draft.length}/{MAX_COMMENT_LEN}
   412→          </p>
   413→        )}
   414→      </div>
   415→    </section>
   416→  )
   417→}
