    83→function computeAvailability(contact: LeadDTO["contact"]): ChannelAvailability[] {
    84→  const defs: { channel: ChannelType; destination: string | null; missingLabel: string }[] = [
    85→    { channel: "whatsapp", destination: contact.phone ? formatPhoneDisplay(contact.phone) : null, missingLabel: "belum ada nomor WhatsApp" },
    86→    { channel: "email", destination: contact.email ?? null, missingLabel: "belum ada email" },
    87→    { channel: "instagram", destination: contact.igUsername ? `@${contact.igUsername}` : null, missingLabel: "belum ada username Instagram" },
    88→  ];
    89→  return defs.map((d) => ({
    90→    channel: d.channel,
    91→    available: Boolean(d.destination),
    92→    destination: d.destination,
    93→    missingLabel: d.destination ? null : d.missingLabel,
    94→  }));
    95→}

    97→function timeAgo(iso: string): string {
    98→  const diff = Date.now() - new Date(iso).getTime();
    99→  const m = Math.floor(diff / 60000);
   100→  if (m < 1) return "baru saja";
   101→  if (m < 60) return `${m}m`;
   102→  const h = Math.floor(m / 60);
   103→  if (h < 24) return `${h}j`;
   104→  return `${Math.floor(h / 24)}h`;
   105→}

   343→      toast.error((e as Error).message);

   345→      setSending(false);
