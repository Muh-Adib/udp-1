    69→    69→function dueInfo(task: TaskDTO): { label: string; overdue: boolean } {
    70→    70→  if (!task.dueDate) return { label: 'Tanpa tenggat', overdue: false }
    71→    71→  const d = daysUntil(task.dueDate)
    72→    72→  if (d === null) return { label: formatDate(task.dueDate), overdue: false }
    73→    73→  if (d < 0) return { label: `${formatDate(task.dueDate)} · terlambat ${Math.abs(d)} hari`, overdue: true }
    74→    74→  if (d === 0) return { label: `${formatDate(task.dueDate)} · hari ini`, overdue: false }
    75→    75→  return { label: `${formatDate(task.dueDate)} · ${d} hari lagi`, overdue: false }
    76→    76→}
