     9→} from '@/components/ui/sheet'

    11→import { Button } from '@/components/ui/button'

    14→import { Checkbox } from '@/components/ui/checkbox'
    15→import { Progress } from '@/components/ui/progress'
    16→import { Alert, AlertDescription, AlertTitle } from '@/components/ui/alert'
    17→import { Skeleton } from '@/components/ui/skeleton'
    18→import { Separator } from '@/components/ui/separator'
    19→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    20→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    21→import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from '@/components/ui/dialog'

    29→import type { OpportunityDetailDTO, Stage, Temperature, Priority, ContactDTO } from '@/lib/crm-types'
    30→import {
    31→  STAGES, stageMeta, LOST_REASONS, lostReasonLabel, REACTIVATION_OPTIONS, reactivationLabel,
    32→  channelMeta, LEAD_SOURCES, leadSourceLabel, PRIORITIES, priorityMeta, TEMPERATURES, temperatureMeta,
    33→  TASK_TYPES, projectStatusMeta, formatMoney, formatDate, timeAgo,
    34→} from '@/lib/crm-constants'
    35→import {
    36→  Building2, User, Mail, MessageCircle, MoreVertical, AlertTriangle, CheckCircle2, Circle,
    37→  Clock, Send, Trash2, Settings2, CalendarClock, Paperclip, Link2, CornerDownRight, X, Calculator,
    38→} from 'lucide-react'

    40→const SCROLLBAR = '[&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-slate-300'
    41→
    42→type TabKey = 'ringkasan' | 'percakapan' | 'tugas' | 'catatan' | 'brief_estimasi'
    43→
    44→interface LostForm { reason: string; notes: string; competitor: string; reactivation: string; followUpDate: string }
    45→
    46→const EMPTY_LOST: LostForm = { reason: '', notes: '', competitor: '', reactivation: '', followUpDate: '' }
    47→
    48→/* ---------- Small definition item ---------- */
    49→function DefItem({ label, value }: { label: string; value?: React.ReactNode | null }) {
    50→  const empty = value === null || value === undefined || value === ''
    51→  return (
    52→    <div className="min-w-0">
    53→      <p className="text-[11px] font-semibold uppercase tracking-wide text-slate-400">{label}</p>
    54→      <div className={cn('mt-0.5 whitespace-pre-wrap break-words text-sm', empty ? 'text-slate-300' : 'text-slate-700')}>
    55→        {empty ? '—' : value}
    56→      </div>
    57→    </div>
    58→  )
    59→}
    60→
    61→export function OpportunityDetailDrawer({ opportunityId, open, onClose, onChanged }: {
    62→  opportunityId: string | null
    63→  open: boolean
    64→  onClose: () => void
    65→  onChanged?: () => void
    66→}) {
    67→  const { toast } = useToast()
    68→  const { user, openOpportunity } = useCrmStore()
    69→
    70→  const [detail, setDetail] = useState<OpportunityDetailDTO | null>(null)
    71→  const [contact, setContact] = useState<ContactDTO | null>(null)
    72→  const [loading, setLoading] = useState(false)
    73→  const [error, setError] = useState<string | null>(null)
    74→  const [tab, setTab] = useState<TabKey>('ringkasan')
    75→
    76→  /* composer */
    77→  const composerRef = useRef<HTMLDivElement | null>(null)
    78→  const [composerChannel, setComposerChannel] = useState('WHATSAPP')
    79→  const [composerBody, setComposerBody] = useState('')
    80→  const [sending, setSending] = useState(false)
    81→
    82→  /* task form */
    83→  const [taskTitle, setTaskTitle] = useState('')
    84→  const [taskAssignee, setTaskAssignee] = useState('')
    85→  const [taskDue, setTaskDue] = useState('')
    86→  const [taskType, setTaskType] = useState('FOLLOW_UP')
    87→  const [addingTask, setAddingTask] = useState(false)
    88→  const [togglingTask, setTogglingTask] = useState<string | null>(null)
    89→
    90→  /* note form */
    91→  const [noteBody, setNoteBody] = useState('')
    92→  const [noteVisibility, setNoteVisibility] = useState('INTERNAL')
    93→  const [addingNote, setAddingNote] = useState(false)
    94→
    95→  /* dialogs */
    96→  const [movingStage, setMovingStage] = useState(false)
    97→  const [lostOpen, setLostOpen] = useState(false)
    98→  const [lostForm, setLostForm] = useState<LostForm>(EMPTY_LOST)
    99→  const [wonOpen, setWonOpen] = useState(false)
   100→  const [wonOffer, setWonOffer] = useState('')
   101→  const [priorityOpen, setPriorityOpen] = useState(false)
   102→  const [priPriority, setPriPriority] = useState<Priority>('MEDIUM')
   103→  const [priTemperature, setPriTemperature] = useState<Temperature>('WARM')
   104→  const [nextActionOpen, setNextActionOpen] = useState(false)
   105→  const [naText, setNaText] = useState('')
   106→  const [naDate, setNaDate] = useState('')
   107→  const [deleteOpen, setDeleteOpen] = useState(false)
   108→  const [deleting, setDeleting] = useState(false)
   109→
   110→  const users = useCrmStore((s) => s.users)
   111→  const activeUsers = useMemo(() => users.filter((u) => u.isActive), [users])
   112→
   113→  /* ---------- data ---------- */
   114→  const load = useCallback(async () => {
   115→    if (!opportunityId) return
   116→    setLoading(true)
   117→    setError(null)
   118→    try {
   119→      const d = await crmApi.opportunity(opportunityId)
   120→      setDetail(d)
   121→      try {
   122→        const cs = await crmApi.contacts(`companyId=${d.companyId}`)
   123→        setContact(cs.find((c) => c.id === d.contactId) ?? null)
   124→      } catch { setContact(null) }
   125→    } catch (e) {
   126→      setError(e instanceof Error ? e.message : 'Gagal memuat detail opportunity')
   127→    } finally {
   128→      setLoading(false)
   129→    }
   130→  }, [opportunityId])
   131→
   132→  useEffect(() => {
   133→    if (open && opportunityId) {
   134→      void load()
   135→    }
   136→    if (!open) {
   137→      setDetail(null)
   138→      setContact(null)
   139→      setError(null)
   140→      setTab('ringkasan')
   141→      setComposerBody('')
   142→      setComposerChannel('WHATSAPP')
   143→      setTaskTitle(''); setTaskDue(''); setTaskType('FOLLOW_UP'); setTaskAssignee('')
   144→      setNoteBody(''); setNoteVisibility('INTERNAL')
   145→      setLostForm(EMPTY_LOST); setWonOffer('')
   146→    }
   147→  }, [open, opportunityId, load])
   148→
   149→  /* ---------- stage mutations ---------- */
   150→  const doChangeStage = useCallback(async (stage: Stage, extra?: Record<string, unknown>, successTitle?: string) => {
   151→    if (!detail) return
   152→    setMovingStage(true)
   153→    try {
   154→      await crmApi.changeStage(detail.id, { stage, ...extra })
   155→      toast({ title: successTitle ?? 'Stage diperbarui', description: `${detail.code} dipindah ke ${stageMeta(stage).label}` })
   156→      setLostOpen(false); setWonOpen(false)
   157→      await load()
   158→      onChanged?.()
   159→    } catch (e) {
   160→      toast({ title: 'Gagal memindah stage', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   161→    } finally {
   162→      setMovingStage(false)
   163→    }
   164→  }, [detail, load, onChanged, toast])
   165→
   166→  const handleStageIntent = (target: string) => {
   167→    if (!detail || target === detail.stage) return
   168→    if (target === 'LOST') { setLostForm(EMPTY_LOST); setLostOpen(true); return }
   169→    if (target === 'WON') { setWonOffer(''); setWonOpen(true); return }
   170→    void doChangeStage(target as Stage)
   171→  }
   172→
   173→  const submitLost = () => {
   174→    if (!lostForm.reason) return
   175→    void doChangeStage('LOST', {
   176→      lostReason: lostForm.reason,
   177→      ...(lostForm.notes.trim() ? { lostNotes: lostForm.notes.trim() } : {}),
   178→      ...(lostForm.competitor.trim() ? { competitorName: lostForm.competitor.trim() } : {}),
   179→      ...(lostForm.reactivation ? { reactivation: lostForm.reactivation } : {}),
   180→      ...(lostForm.followUpDate ? { followUpDate: lostForm.followUpDate } : {}),
   181→    }, 'Opportunity ditandai Lost')
   182→  }
   183→
   184→  const confirmWon = () => {
   185→    const offer = Number(wonOffer)
   186→    void doChangeStage('WON', offer > 0 ? { lastOfferValue: offer } : {}, '🎉 Deal Won! Project produksi dibuat otomatis')
   187→  }
   188→
   189→  /* ---------- priority / next action / delete ---------- */
   190→  const openPriorityDialog = () => {
   191→    if (!detail) return
   192→    setPriPriority(detail.priority)
   193→    setPriTemperature(detail.temperature)
   194→    setPriorityOpen(true)
   195→  }
   196→
   197→  const savePriority = async () => {
   198→    if (!detail) return
   199→    if (priPriority === detail.priority && priTemperature === detail.temperature) { setPriorityOpen(false); return }
   200→    try {
   201→      await crmApi.updateOpportunity(detail.id, { priority: priPriority, temperature: priTemperature })
   202→      toast({ title: 'Prioritas diperbarui', description: `${detail.code} → ${priorityMeta(priPriority).label} · ${temperatureMeta(priTemperature).label}` })
   203→      setPriorityOpen(false)
   204→      await load()
   205→      onChanged?.()
   206→    } catch (e) {
   207→      toast({ title: 'Gagal memperbarui prioritas', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   208→    }
   209→  }
   210→
   211→  const openNextActionDialog = () => {
   212→    if (!detail) return
   213→    setNaText(detail.nextAction ?? '')
   214→    setNaDate(detail.nextActionDate ? detail.nextActionDate.slice(0, 10) : '')
   215→    setNextActionOpen(true)
   216→  }
   217→
   218→  const saveNextAction = async () => {
   219→    if (!detail) return
   220→    const text = naText.trim()
   221→    const same = text === (detail.nextAction ?? '') && naDate === (detail.nextActionDate ? detail.nextActionDate.slice(0, 10) : '')
   222→    if (same) { setNextActionOpen(false); return }
   223→    try {
   224→      await crmApi.updateOpportunity(detail.id, { nextAction: text, nextActionDate: naDate })
   225→      toast({ title: 'Next action diperbarui' })
   226→      setNextActionOpen(false)
   227→      await load()
   228→      onChanged?.()
   229→    } catch (e) {
   230→      toast({ title: 'Gagal memperbarui next action', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   231→    }
   232→  }
   233→
   234→  const confirmDelete = async () => {
   235→    if (!detail) return
   236→    setDeleting(true)
   237→    try {
   238→      await crmApi.deleteOpportunity(detail.id)
   239→      toast({ title: 'Opportunity dihapus', description: `${detail.code} dihapus dari pipeline` })
   240→      setDeleteOpen(false)
   241→      onClose()
   242→      onChanged?.()
   243→    } catch (e) {
   244→      toast({ title: 'Gagal menghapus', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   245→    } finally {
   246→      setDeleting(false)
   247→    }
   248→  }
   249→
   250→  /* ---------- interactions ---------- */
   251→  const scrollToComposer = () => {
   252→    setTab('percakapan')
   253→    setTimeout(() => composerRef.current?.scrollIntoView({ behavior: 'smooth', block: 'center' }), 80)
   254→  }
   255→
   256→  const sendInteraction = async () => {
   257→    if (!detail || !composerBody.trim()) return
   258→    setSending(true)
   259→    try {
   260→      await crmApi.addInteraction(detail.id, { channel: composerChannel, direction: 'OUT', body: composerBody.trim() })
   261→      toast({ title: 'Terkirim — stage otomatis naik ke Connected bila sebelumnya New' })
   262→      setComposerBody('')
   263→      await load()
   264→      onChanged?.()
   265→    } catch (e) {
   266→      toast({ title: 'Gagal mengirim', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   267→    } finally {
   268→      setSending(false)
   269→    }
   270→  }
   271→
   272→  /* ---------- tasks ---------- */
   273→  const toggleTask = async (taskId: string, done: boolean) => {
   274→    setTogglingTask(taskId)
   275→    try {
   276→      await crmApi.updateTask(taskId, { status: done ? 'DONE' : 'OPEN' })
   277→      await load()
   278→      onChanged?.()
   279→    } catch (e) {
   280→      toast({ title: 'Gagal memperbarui task', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   281→    } finally {
   282→      setTogglingTask(null)
   283→    }
   284→  }
   285→
   286→  const submitTask = async () => {
   287→    if (!detail) return
   288→    if (!taskTitle.trim() || !taskAssignee) {
   289→      toast({ title: 'Lengkapi judul dan assignee task', variant: 'destructive' })
   290→      return
   291→    }
   292→    setAddingTask(true)
   293→    try {
   294→      await crmApi.createTask({
   295→        opportunityId: detail.id,
   296→        title: taskTitle.trim(),
   297→        assigneeId: taskAssignee,
   298→        type: taskType,
   299→        ...(taskDue ? { dueDate: taskDue } : {}),
   300→      })
   301→      toast({ title: 'Task ditambahkan' })
   302→      setTaskTitle(''); setTaskDue(''); setTaskType('FOLLOW_UP')
   303→      await load()
   304→      onChanged?.()
   305→    } catch (e) {
   306→      toast({ title: 'Gagal menambah task', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   307→    } finally {
   308→      setAddingTask(false)
   309→    }
   310→  }
   311→
   312→  /* ---------- notes ---------- */
   313→  const submitNote = async () => {
   314→    if (!detail || !noteBody.trim()) return
   315→    setAddingNote(true)
   316→    try {
   317→      await crmApi.addNote(detail.id, { body: noteBody.trim(), visibility: noteVisibility })
   318→      toast({ title: 'Catatan disimpan' })
   319→      setNoteBody('')
   320→      await load()
   321→      onChanged?.()
   322→    } catch (e) {
   323→      toast({ title: 'Gagal menyimpan catatan', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   324→    } finally {
   325→      setAddingNote(false)
   326→    }
   327→  }
   328→
   329→  /* ---------- derived ---------- */
   330→  const overdueTasks = useMemo(() => {
   331→    if (!detail) return []
   332→    const now = Date.now()
   333→    return detail.tasks.filter((t) => t.status !== 'DONE' && t.dueDate && new Date(t.dueDate).getTime() < now)
   334→  }, [detail])
   335→
   336→  /* ---------- render ---------- */
   337→  return (
   338→    <Sheet open={open} onOpenChange={(o) => { if (!o) onClose() }}>
   339→      <SheetContent side="right" className={cn('flex w-full flex-col gap-0 overflow-y-auto p-0 sm:max-w-2xl [&>button]:hidden', SCROLLBAR)}>
   340→        <SheetTitle className="sr-only">Detail Opportunity</SheetTitle>
   341→        <SheetDescription className="sr-only">Ringkasan lengkap opportunity, percakapan, tugas, dan catatan</SheetDescription>
   342→
   343→        {/* loading / error / empty states */}
   344→        {loading && !detail && (
   345→          <div className="space-y-4 p-5">
   346→            <Skeleton className="h-4 w-24" />
   347→            <Skeleton className="h-7 w-3/4" />
   348→            <div className="flex gap-2"><Skeleton className="h-6 w-20" /><Skeleton className="h-6 w-24" /><Skeleton className="h-6 w-16" /></div>
   349→            <Skeleton className="h-8 w-56" />
   350→            <Skeleton className="h-24 w-full rounded-xl" />
   351→            <Skeleton className="h-64 w-full rounded-xl" />
   352→          </div>
   353→        )}
   354→
   355→        {!loading && error && (
   356→          <div className="p-5">
   357→            <Alert variant="destructive">
   358→              <AlertTriangle className="h-4 w-4" />
   359→              <AlertTitle>Gagal memuat opportunity</AlertTitle>
   360→              <AlertDescription>{error}</AlertDescription>
   361→            </Alert>
   362→            <Button variant="outline" size="sm" className="mt-3" onClick={() => void load()}>Coba lagi</Button>
   363→          </div>
   364→        )}
   365→
   366→        {detail && (
   367→          <>
   368→            {/* ---------- sticky header ---------- */}
   369→            <div className="sticky top-0 z-20 border-b border-slate-100 bg-white/95 px-5 pb-4 pt-4 backdrop-blur">
   370→              <div className="flex items-start justify-between gap-2 pr-8">
   371→                <div className="min-w-0">
   372→                  <p className="font-mono text-xs text-slate-400">{detail.code}</p>
   373→                  <h2 className="mt-0.5 text-lg font-bold leading-snug text-slate-900">{detail.title}</h2>
   374→                </div>
   375→                <SheetClose asChild>
   376→                  <Button variant="ghost" size="icon" className="h-8 w-8 shrink-0 text-slate-400 hover:text-slate-700" title="Tutup">
   377→                    <X className="h-4 w-4" />
   378→                  </Button>
   379→                </SheetClose>
   380→              </div>
   381→
   382→              <div className="mt-2 flex flex-wrap items-center gap-2">
   383→                <BrandChip name={detail.brandName} color={detail.brandColor} />
   384→                <StageBadge stage={detail.stage} />
   385→                <TempBadge temperature={detail.temperature} />
   386→                <PriorityBadge priority={detail.priority} />
   387→                <span className="rounded-md bg-slate-100 px-2 py-0.5 text-[11px] font-bold text-slate-600">{detail.probability}%</span>
   388→              </div>
   389→
   390→              <div className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-slate-500">
   391→                <span className="inline-flex items-center gap-1.5"><Building2 className="h-3.5 w-3.5 text-slate-400" />{detail.companyName}</span>
   392→                <span className="inline-flex items-center gap-1.5"><User className="h-3.5 w-3.5 text-slate-400" />{detail.contactName}</span>
   393→                {contact?.whatsapp && (
   394→                  <span className="inline-flex items-center gap-1 text-[11px] text-slate-400"><MessageCircle className="h-3 w-3" />{contact.whatsapp}</span>
   395→                )}
   396→                {contact?.email && (
   397→                  <span className="inline-flex items-center gap-1 text-[11px] text-slate-400"><Mail className="h-3 w-3" />{contact.email}</span>
   398→                )}
   399→              </div>
   400→
   401→              <div className="mt-3 flex flex-wrap items-end justify-between gap-3">
   402→                <div>
   403→                  <p className="text-xl font-extrabold tracking-tight text-slate-900">{formatMoney(detail.estimatedValue, detail.currency)}</p>
   404→                  <p className="inline-flex items-center gap-1 text-[11px] text-slate-400"><CalendarClock className="h-3 w-3" />Est. close: {formatDate(detail.expectedCloseDate)}</p>
   405→                </div>
   406→                <div className="flex flex-wrap items-center gap-2">
   407→                  <Button variant="outline" size="sm" className="gap-1.5" onClick={scrollToComposer}>✉️ Log Percakapan</Button>
   408→                  <Select value={detail.stage} onValueChange={handleStageIntent} disabled={movingStage}>
   409→                    <SelectTrigger size="sm" className="w-[168px]" title="Pindah stage">
   410→                      <SelectValue />
   411→                    </SelectTrigger>
   412→                    <SelectContent>
   413→                      {STAGES.map((s) => (
   414→                        <SelectItem key={s.key} value={s.key} disabled={s.key === detail.stage}>
   415→                          <span className="inline-flex items-center gap-2"><span className={cn('h-1.5 w-1.5 rounded-full', s.dot)} />{s.label}</span>
   416→                        </SelectItem>
   417→                      ))}
   418→                    </SelectContent>
   419→                  </Select>
   420→                  <DropdownMenu>
   421→                    <DropdownMenuTrigger asChild>
   422→                      <Button variant="ghost" size="icon" className="h-8 w-8 text-slate-500" title="Aksi lainnya">
   423→                        <MoreVertical className="h-4 w-4" />
   424→                      </Button>
   425→                    </DropdownMenuTrigger>
   426→                    <DropdownMenuContent align="end">
   427→                      <DropdownMenuItem onClick={openPriorityDialog}><Settings2 className="mr-2 h-4 w-4" />Edit Prioritas</DropdownMenuItem>
   428→                      <DropdownMenuItem onClick={openNextActionDialog}><CalendarClock className="mr-2 h-4 w-4" />Edit Next Action</DropdownMenuItem>
   429→                      <DropdownMenuSeparator />
   430→                      <DropdownMenuItem className="text-rose-600 focus:text-rose-700" onClick={() => setDeleteOpen(true)}>
   431→                        <Trash2 className="mr-2 h-4 w-4" />Hapus
   432→                      </DropdownMenuItem>
   433→                    </DropdownMenuContent>
   434→                  </DropdownMenu>
   435→                </div>
   436→              </div>
   437→            </div>
   438→
   439→            {/* ---------- tabs ---------- */}
   440→            <Tabs value={tab} onValueChange={(v) => setTab(v as TabKey)} className="flex min-h-0 flex-1 flex-col">
   441→              <div className="sticky top-0 z-10 bg-white px-5 pt-3">
   442→                <TabsList className="w-full justify-start overflow-x-auto">
   443→                  <TabsTrigger value="ringkasan">Ringkasan</TabsTrigger>
   444→                  <TabsTrigger value="percakapan" className="gap-1.5">Percakapan <span className="rounded-full bg-slate-100 px-1.5 text-[10px] font-semibold text-slate-500">{detail.interactions.length}</span></TabsTrigger>
   445→                  <TabsTrigger value="tugas" className="gap-1.5">Tugas <span className="rounded-full bg-slate-100 px-1.5 text-[10px] font-semibold text-slate-500">{detail.tasks.length}</span></TabsTrigger>
   446→                  <TabsTrigger value="catatan" className="gap-1.5">Catatan <span className="rounded-full bg-slate-100 px-1.5 text-[10px] font-semibold text-slate-500">{detail.notes.length}</span></TabsTrigger>
   447→                  <TabsTrigger value="brief_estimasi" className="gap-1.5"><Calculator className="h-3.5 w-3.5" /> Brief &amp; Estimasi</TabsTrigger>
   448→                </TabsList>
   449→              </div>
