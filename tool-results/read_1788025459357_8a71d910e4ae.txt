    70→function dueInfo(task: TaskDTO): { label: string; overdue: boolean } {
    71→  if (!task.dueDate) return { label: 'Tanpa tenggat', overdue: false }
    72→  const d = daysUntil(task.dueDate)
    73→  if (d === null) return { label: formatDate(task.dueDate), overdue: false }
    74→  if (d < 0) return { label: `${formatDate(task.dueDate)} · terlambat ${Math.abs(d)} hari`, overdue: true }
    75→  if (d === 0) return { label: `${formatDate(task.dueDate)} · hari ini`, overdue: false }
    76→  return { label: `${formatDate(task.dueDate)} · ${d} hari lagi`, overdue: false }
    77→}
