    44→/* ---------------- helper format lokal ---------------- */
    45→
    46→const rpFormatter = new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", maximumFractionDigits: 0 });
    47→
    48→function formatRp(n: number): string {
    49→  return rpFormatter.format(Number.isFinite(n) ? Math.round(n) : 0);
    50→}
    51→
    52→function formatDate(iso?: string | null): string {
    53→  if (!iso) return "—";
    54→  return new Date(iso).toLocaleDateString("id-ID", { day: "numeric", month: "short", year: "numeric" });
    55→}

    74→export default function FinanceView({ user }: { user: SessionUser }) {
    75→  const [stats, setStats] = useState<FinanceStats | null>(null);
    76→  const [quotations, setQuotations] = useState<QuotationDTO[] | null>(null);
    77→  const [invoices, setInvoices] = useState<InvoiceDTO[] | null>(null);
    78→  const [loading, setLoading] = useState(true);
    79→  const [error, setError] = useState<string | null>(null);
    80→  const [busy, setBusy] = useState<string | null>(null);
    81→
    82→  // dialog tolak penawaran
    83→  const [rejectTarget, setRejectTarget] = useState<QuotationDTO | null>(null);
    84→  const [rejectNote, setRejectNote] = useState("");
    85→
    86→  // dialog buat penawaran
    87→  const [createOpen, setCreateOpen] = useState(false);
    88→  const [leadsCache, setLeadsCache] = useState<LeadDTO[] | null>(null);
    89→  const [leadsLoading, setLeadsLoading] = useState(false);
    90→  const [briefsCache, setBriefsCache] = useState<BriefDTO[] | null>(null);
    91→  const [formLeadId, setFormLeadId] = useState("");
    92→  const [formTitle, setFormTitle] = useState("");
    93→  const [formItems, setFormItems] = useState<DraftItem[]>([emptyDraftItem()]);
    94→  const [formDiscount, setFormDiscount] = useState("0");
    95→  const [formPpn, setFormPpn] = useState("11");
    96→  const [formNotes, setFormNotes] = useState("");
    97→
    98→  // dialog catat pembayaran
    99→  const [payTarget, setPayTarget] = useState<InvoiceDTO | null>(null);
   100→  const [payAmount, setPayAmount] = useState("0");
   101→  const [payMethod, setPayMethod] = useState("TRANSFER");
   102→  const [payNote, setPayNote] = useState("");
   103→
   104→  const role = user.role;
   105→  const canCreate = role === "OWNER" || role === "MANAGER" || role === "MARKETER";
   106→  const canSend = canCreate;
   107→  const canDecide = role === "OWNER" || role === "MANAGER" || role === "FINANCE";
   108→  const canPay = role === "OWNER" || role === "FINANCE";
   109→
   110→  /* ---------- load paralel saat mount ---------- */
   111→
   112→  const load = useCallback(async () => {
   113→    setLoading(true);
   114→    setError(null);
   115→    try {
   116→      const [s, q, i] = await Promise.all([api.financeStats(), api.quotations(), api.invoices()]);
   117→      setStats(s.stats);
   118→      setQuotations(q.quotations);
   119→      setInvoices(i.invoices);
   120→    } catch (e) {
   121→      const msg = (e as Error).message || "Gagal memuat data keuangan";
   122→      setError(msg);
   123→      if (stats || quotations || invoices) toast.error(msg);
   124→    } finally {
   125→      setLoading(false);
   126→    }
   127→    // eslint-disable-next-line react-hooks/exhaustive-deps
   128→  }, []);
   129→
   130→  useEffect(() => {
   131→    void load();
   132→  }, [load]);
   133→
   134→  /** Refetch ringan setelah aksi (quotations + invoices + stats) tanpa skeleton penuh. */
   135→  const refreshData = useCallback(async () => {
   136→    try {
   137→      const [s, q, i] = await Promise.all([api.financeStats(), api.quotations(), api.invoices()]);
   138→      setStats(s.stats);
   139→      setQuotations(q.quotations);
   140→      setInvoices(i.invoices);
   141→    } catch (e) {
   142→      toast.error((e as Error).message || "Gagal memuat ulang data");
   143→    }
   144→  }, []);
   145→
   146→  /* ---------- aksi penawaran ---------- */
   147→
   148→  async function sendQuotation(q: QuotationDTO) {
   149→    setBusy(`send-${q.id}`);
   150→    try {
   151→      await api.updateQuotationStatus(q.id, "send");
   152→      toast.success(`Penawaran ${q.number} dikirim ke klien`);
   153→      await refreshData();
   154→    } catch (e) {
   155→      toast.error((e as Error).message);
   156→    } finally {
   157→      setBusy(null);
   158→    }
   159→  }
   160→
   161→  async function approveQuotation(q: QuotationDTO) {
   162→    setBusy(`approve-${q.id}`);
   163→    try {
   164→      const r = await api.updateQuotationStatus(q.id, "approve");
   165→      if (r.projectCode && r.invoiceNumber) {
   166→        toast.success(`Penawaran disetujui — Proyek ${r.projectCode} & invoice ${r.invoiceNumber} otomatis dibuat`);
   167→      } else {
   168→        toast.success(`Penawaran ${q.number} disetujui`);
   169→      }
   170→      await refreshData();
   171→    } catch (e) {
   172→      toast.error((e as Error).message);
   173→    } finally {
   174→      setBusy(null);
   175→    }
   176→  }
   177→
   178→  function openReject(q: QuotationDTO) {
   179→    setRejectTarget(q);
   180→    setRejectNote("");
   181→  }
   182→
   183→  async function submitReject() {
   184→    if (!rejectTarget) return;
   185→    setBusy("reject");
   186→    try {
   187→      await api.updateQuotationStatus(rejectTarget.id, "reject", rejectNote.trim() || undefined);
   188→      toast.success(`Penawaran ${rejectTarget.number} ditolak`);
   189→      setRejectTarget(null);
   190→      await refreshData();
   191→    } catch (e) {
   192→      toast.error((e as Error).message);
   193→    } finally {
   194→      setBusy(null);
   195→    }
   196→  }
   197→
   198→  /* ---------- dialog buat penawaran ---------- */
   199→
   200→  function openCreate() {
   201→    setCreateOpen(true);
   202→    // fetch sekali saat dialog pertama kali dibuka (lead + brief utk referensi estimasi)
   203→    if (leadsCache === null) {
   204→      setLeadsLoading(true);
   205→      api
   206→        .leads()
   207→        .then(({ leads }) => setLeadsCache(leads))
   208→        .catch((e) => toast.error((e as Error).message))
   209→        .finally(() => setLeadsLoading(false));
   210→    }
   211→    if (briefsCache === null) {
   212→      api
   213→        .briefs()
   214→        .then(({ briefs }) => setBriefsCache(briefs))
   215→        .catch(() => setBriefsCache([]));
   216→    }
   217→  }
   218→
   219→  function resetCreateForm() {
   220→    setFormLeadId("");
   221→    setFormTitle("");
   222→    setFormItems([emptyDraftItem()]);
   223→    setFormDiscount("0");
   224→    setFormPpn("11");
   225→    setFormNotes("");
   226→  }
   227→
   228→  const formSubtotal = formItems.reduce((s, it) => s + (Number(it.qty) || 0) * (Number(it.price) || 0), 0);
   229→  const formDiscountVal = formSubtotal * ((Number(formDiscount) || 0) / 100);
   230→  const formAfterDiscount = formSubtotal - formDiscountVal;
   231→  const formPpnVal = formAfterDiscount * ((Number(formPpn) || 0) / 100);
   232→  const formGrandTotal = formAfterDiscount + formPpnVal;
   233→
   234→  async function submitCreate() {
   235→    if (!formLeadId) {
   236→      toast.error("Pilih lead terlebih dahulu");
   237→      return;
   238→    }
   239→    const title = formTitle.trim();
   240→    if (!title) {
   241→      toast.error("Judul penawaran wajib diisi");
   242→      return;
   243→    }
   244→    const items = formItems.map((it) => ({
   245→      desc: it.desc.trim(),
   246→      qty: Math.max(0, Math.floor(Number(it.qty) || 0)),
   247→      price: Math.max(0, Math.round(Number(it.price) || 0)),
   248→    }));
   249→    if (items.some((it) => !it.desc)) {
   250→      toast.error("Deskripsi setiap item wajib diisi");
   251→      return;
   252→    }
   253→    if (items.some((it) => it.qty < 1)) {
   254→      toast.error("Qty setiap item minimal 1");
   255→      return;
   256→    }
   257→    if (formSubtotal <= 0) {
   258→      toast.error("Total penawaran harus lebih dari 0");
   259→      return;
   260→    }
   261→    setBusy("create");
   262→    try {
   263→      const { quotation } = await api.createQuotation({
   264→        leadId: formLeadId,
   265→        title,
   266→        items,
   267→        discountPct: Number(formDiscount) || 0,
   268→        ppnPct: Number(formPpn) || 0,
   269→        notes: formNotes.trim() || undefined,
   270→      });
   271→      toast.success(`Penawaran ${quotation.number} dibuat (draf)`);
   272→      setCreateOpen(false);
   273→      resetCreateForm();
   274→      await refreshData();
   275→    } catch (e) {
   276→      toast.error((e as Error).message);
   277→    } finally {
   278→      setBusy(null);
   279→    }
   280→  }
   281→
   282→  /* ---------- dialog catat pembayaran ---------- */
   283→
   284→  function openPay(inv: InvoiceDTO) {
   285→    setPayTarget(inv);
   286→    setPayAmount(String(Math.max(0, inv.grandTotal - inv.paidAmount)));
   287→    setPayMethod("TRANSFER");
   288→    setPayNote("");
   289→  }
   290→
   291→  async function submitPay() {
   292→    if (!payTarget) return;
   293→    const amount = Math.round(Number(payAmount) || 0);
   294→    const remaining = payTarget.grandTotal - payTarget.paidAmount;
   295→    if (amount <= 0) {
   296→      toast.error("Nominal pembayaran tidak valid");
   297→      return;
   298→    }
   299→    if (amount > remaining) {
   300→      toast.error(`Nominal melebihi sisa tagihan (${formatRp(remaining)})`);
   301→      return;
   302→    }
   303→    setBusy(`pay-${payTarget.id}`);
   304→    try {
   305→      await api.addPayment(payTarget.id, { amount, method: payMethod, note: payNote.trim() || undefined });
   306→      toast.success(`Pembayaran ${formatRp(amount)} untuk ${payTarget.number} tercatat`);
   307→      setPayTarget(null);
   308→      await refreshData();
   309→    } catch (e) {
   310→      toast.error((e as Error).message);
   311→    } finally {
   312→      setBusy(null);
   313→    }
   314→  }
   315→
   316→  /* ---------- render: error ---------- */
   317→
   318→  if (error && (!stats || !quotations || !invoices)) {

   321→        <CardContent className="flex flex-col items-center gap-3 py-10 text-center">

   323→            <TriangleAlert className="size-6" />
   324→          </span>
   325→          <div>
   326→            <p className="font-semibold">Gagal memuat data keuangan</p>
   327→            <p className="text-sm text-muted-foreground">{error}</p>
   328→          </div>
   329→          <Button onClick={() => void load()}>
   330→            <RefreshCw className="size-4" /> Coba Lagi
   331→          </Button>
   332→        </CardContent>
   333→      </Card>
   334→    );
   335→  }
   336→
   337→  /* ---------- render: skeleton ---------- */
   338→
   339→  if (loading || !stats || !quotations || !invoices) {
   340→    return (
   341→      <div className="space-y-5">
   342→        <div className="animate-pulse space-y-2">
   343→          <div className="h-6 w-40 rounded bg-muted" />
   344→          <div className="h-3 w-72 rounded bg-muted" />
   345→        </div>
   346→        <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
   347→          {[0, 1, 2, 3].map((i) => (
   348→            <Card key={i}>
   349→              <CardContent className="animate-pulse space-y-3 pt-5">
   350→                <div className="h-3 w-24 rounded bg-muted" />
   351→                <div className="h-7 w-36 rounded bg-muted" />
   352→                <div className="h-3 w-28 rounded bg-muted" />
   353→              </CardContent>
   354→            </Card>
   355→          ))}
   356→        </div>
   357→        <div className="grid gap-4 lg:grid-cols-5">
   358→          <Card className="lg:col-span-3">
   359→            <CardContent className="animate-pulse space-y-3 pt-5">
   360→              <div className="h-4 w-44 rounded bg-muted" />
   361→              <div className="h-40 rounded-xl bg-muted" />
   362→            </CardContent>
   363→          </Card>
   364→          <Card className="lg:col-span-2">
   365→            <CardContent className="animate-pulse space-y-3 pt-5">
   366→              <div className="h-4 w-40 rounded bg-muted" />
   367→              {[0, 1, 2].map((i) => (
   368→                <div key={i} className="h-8 rounded-lg bg-muted" />
   369→              ))}
   370→            </CardContent>
   371→          </Card>
   372→        </div>
   373→        <Card>
   374→          <CardContent className="animate-pulse space-y-3 pt-5">
   375→            <div className="h-9 w-56 rounded-lg bg-muted" />
   376→            {[0, 1, 2, 3, 4].map((i) => (
   377→              <div key={i} className="h-10 rounded-lg bg-muted" />
   378→            ))}
   379→          </CardContent>
   380→        </Card>
   381→      </div>
   382→    );
   383→  }
   384→
   385→  /* ---------- data siap ---------- */
   386→
   387→  const maxCash = Math.max(1, ...stats.monthly.map((m) => Math.max(m.revenue, m.invoiced)));
   388→  const barHeight = (v: number) => {
   389→    const px = Math.round((v / maxCash) * 160);
   390→    return `${Math.max(v > 0 ? 6 : 3, px)}px`;
   391→  };
   392→  const maxBrandRevenue = Math.max(1, ...stats.byBrand.map((b) => b.revenue));
   393→
   394→  const kpis = [
   395→    {
   396→      key: "revenue",
   397→      label: "Pendapatan",
   398→      value: formatRp(stats.revenuePaid),
   399→      sub: "Total pembayaran masuk",
   400→      icon: Banknote,
   401→      iconClass: "bg-emerald-100 text-emerald-700",
   402→      alert: false,
   403→    },
   404→    {
   405→      key: "outstanding",
   406→      label: "Outstanding",
   407→      value: formatRp(stats.outstanding),
   408→      sub: `Belum dibayar dari ${stats.invoiceCount} invoice`,
   409→      icon: Wallet,
   410→      iconClass: "bg-amber-100 text-amber-700",
   411→      alert: false,
   412→    },
   413→    {
   414→      key: "overdue",
   415→      label: "Invoice Jatuh Tempo",
   416→      value: String(stats.overdueCount),
   417→      sub: stats.overdueCount > 0 ? "Segera tindak lanjuti penagihan" : "Semua tagihan terkendali",
   418→      icon: AlarmClock,
   419→      iconClass: stats.overdueCount > 0 ? "bg-rose-100 text-rose-600" : "bg-stone-200 text-stone-600",
   420→      alert: stats.overdueCount > 0,
   421→    },
   422→    {
   423→      key: "approval",
   424→      label: "Tingkat Persetujuan",
   425→      value: `${Math.round(stats.quotationApprovedPct)}%`,
   426→      sub: `Dari ${stats.quotationCount} penawaran`,
   427→      icon: CheckCircle2,
   428→      iconClass: "bg-emerald-100 text-emerald-700",
   429→      alert: false,
   430→    },
   431→  ];
   432→
   433→  return (
   434→    <div className="space-y-5">
   435→      {/* Header */}
   436→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   437→        <div>
   438→          <h2 className="text-xl font-bold tracking-tight">Keuangan</h2>
   439→          <p className="text-sm text-muted-foreground">
   440→            Penawaran, invoice, dan arus kas — gabungan seluruh brand PT. Unicam Digital Pictvres.
   441→          </p>
   442→        </div>
   443→        <Button variant="outline" size="sm" onClick={() => void load()} disabled={loading}>
   444→          <RefreshCw className={cn("size-4", loading && "animate-spin")} /> Muat Ulang

   450→        {kpis.map((k) => {
   451→          const Icon = k.icon;

   454→              <CardContent className="flex items-start justify-between gap-3 pt-5">
   455→                <div className="min-w-0 space-y-1">
   456→                  <p className="text-sm text-muted-foreground">{k.label}</p>

   465→                </span>

   468→          );
   469→        })}
   470→      </div>
   471→
   472→      {/* Charts */}
   473→      <div className="grid gap-4 lg:grid-cols-5">
   474→        {/* Cashflow 6 bulan — grouped bar CSS murni */}
   475→        <Card className="lg:col-span-3">
   476→          <CardHeader className="flex-row items-start justify-between space-y-0">
   477→            <div className="space-y-1">
   478→              <CardTitle className="text-base">Arus Kas 6 Bulan</CardTitle>
   479→              <CardDescription>Pembayaran diterima vs nilai invoice yang ditagihkan</CardDescription>
   480→            </div>
   481→            <div className="flex shrink-0 items-center gap-3 text-xs text-muted-foreground">
   482→              <span className="flex items-center gap-1.5">
   483→                <span className="size-2.5 rounded-sm bg-emerald-500" /> Diterima
   484→              </span>
   485→              <span className="flex items-center gap-1.5">
