    84→    84→function validUntilText(q: PortalQuotationDTO): { text: string; urgent: boolean } {
    85→    85→  if (!q.validUntil) return { text: 'Tanpa masa berlaku', urgent: false }
    86→    86→  const d = daysUntil(q.validUntil)
    87→    87→  if (d === null) return { text: `Berlaku hingga ${formatDate(q.validUntil)}`, urgent: false }
    88→    88→  if (d < 0) return { text: `Berlaku hingga ${formatDate(q.validUntil)} · masa berlaku lewat`, urgent: q.status === 'SENT' }
    89→    89→  if (d === 0) return { text: `Berlaku hingga ${formatDate(q.validUntil)} · berakhir hari ini`, urgent: true }
    90→    90→  return { text: `Berlaku hingga ${formatDate(q.validUntil)} · ${d} hari lagi`, urgent: q.status === 'SENT' && d <= 7 }
    91→    91→}

    93→    93→function dueInfoOf(inv: PortalInvoiceDTO): { text: string; overdue: boolean } {
    94→    94→  if (!inv.dueDate) return { text: 'Tanpa jatuh tempo', overdue: false }
    95→    95→  const remaining = Math.max(0, inv.total - inv.paidAmount)
    96→    96→  const d = daysUntil(inv.dueDate)
    97→    97→  if (d === null) return { text: `Jatuh tempo ${formatDate(inv.dueDate)}`, overdue: false }
    98→    98→  if (d < 0) {
    99→    99→    const overdue = remaining > 0 && inv.status !== 'CANCELLED'
   100→   100→    return {
   101→   101→      text: `Jatuh tempo ${formatDate(inv.dueDate)}${overdue ? ` · terlambat ${Math.abs(d)} hari` : ''}`,
   102→   102→      overdue,
   103→   103→    }
   104→   104→  }
   105→   105→  if (d === 0) return { text: `Jatuh tempo ${formatDate(inv.dueDate)} · hari ini`, overdue: false }
   106→   106→  return { text: `Jatuh tempo ${formatDate(inv.dueDate)} · ${d} hari lagi`, overdue: false }
   107→   107→}
