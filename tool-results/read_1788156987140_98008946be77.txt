     3→

     5→import { crmApi } from './api-client'

     7→import { useToast } from '@/hooks/use-toast'
     8→import { LoadingRows, RefreshButton, SectionHeader, UserAvatar } from './shared'
     9→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    10→import { Badge } from '@/components/ui/badge'
    11→import { Button } from '@/components/ui/button'
    12→import {

    14→} from '@/components/ui/dialog'
    15→import { Input } from '@/components/ui/input'

    17→import {
    18→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    19→} from '@/components/ui/select'
    20→import { Switch } from '@/components/ui/switch'
    21→import { Textarea } from '@/components/ui/textarea'
    22→import { Progress } from '@/components/ui/progress'
    23→import { WORKFLOW_MILESTONES, formatMoney } from '@/lib/crm-constants'
    24→import type { BrandDTO, OpportunityDTO, UserDTO } from '@/lib/crm-types'
    25→import { cn } from '@/lib/utils'
    26→import {
    27→  Building2, Globe, Info, Loader2, Mail, MapPin, Instagram, Phone, Save, Settings2,
    28→  Timer, Users2, Workflow,
    29→} from 'lucide-react'
    30→
    31→const WORKFLOW_LABELS: Record<string, string> = {
    32→  website: 'Website', video: 'Video', animation: 'Animasi', livestream: 'Live Streaming', generic: 'Umum',
    33→}
    34→
    35→const CURRENCIES = ['IDR', 'SGD', 'USD'] as const
    36→const WORKFLOWS = ['website', 'video', 'animation', 'livestream', 'generic'] as const
    37→
    38→const COLOR_PRESETS = ['#10b981', '#0d9488', '#d97706', '#7c3aed', '#e11d48', '#0284c7', '#65a30d', '#57534e']
    39→
    40→interface BrandFormState {
    41→  tagline: string
    42→  description: string
    43→  website: string
    44→  email: string
    45→  phone: string
    46→  instagram: string
    47→  address: string
    48→  color: string
    49→  slaHours: string
    50→  primaryCurrency: string
    51→  invoicePrefix: string
    52→  quotationPrefix: string
    53→  workflowType: string

    56→

    58→  tagline: b.tagline ?? '',
    59→  description: b.description ?? '',
    60→  website: (b.website ?? '').replace(/^https?:\/\//, ''),
    61→  email: b.email ?? '',
    62→  phone: b.phone ?? '',
    63→  instagram: b.instagram ?? '',
    64→  address: b.address ?? '',
    65→  color: b.color,
    66→  slaHours: String(b.slaHours),
    67→  primaryCurrency: b.primaryCurrency,
    68→  invoicePrefix: b.invoicePrefix,
    69→  quotationPrefix: b.quotationPrefix,
    70→  workflowType: b.workflowType,
    71→  isActive: true,
    72→})
    73→
    74→export default function BrandsView() {
    75→  const brands = useCrmStore((s) => s.brands)
    76→  const setBrands = useCrmStore((s) => s.setBrands)
    77→  const user = useCrmStore((s) => s.user)
    78→  const { toast } = useToast()
    79→  const [users, setUsers] = useState<UserDTO[]>([])
    80→  const [opps, setOpps] = useState<OpportunityDTO[]>([])
    81→  const [loading, setLoading] = useState(true)
    82→
    83→  /* Pengaturan brand — hanya SUPER_ADMIN & DIREKTUR (selaras gate API) */
    84→  const canManage = user?.role === 'SUPER_ADMIN' || user?.role === 'DIREKTUR'
    85→  const [editing, setEditing] = useState<BrandDTO | null>(null)
    86→  const [form, setForm] = useState<BrandFormState | null>(null)
    87→  const [saving, setSaving] = useState(false)
    88→
    89→  const load = useCallback(async () => {
    90→    setLoading(true)
    91→    try {
    92→      const [u, o] = await Promise.all([crmApi.users(), crmApi.opportunities()])
    93→      setUsers(u); setOpps(o)
    94→    } catch (e) {
    95→      toast({ title: 'Gagal memuat konfigurasi', description: e instanceof Error ? e.message : undefined, variant: 'destructive' })
    96→    } finally { setLoading(false) }
    97→  }, [toast])
    98→
    99→  useEffect(() => { load() }, [load])
   100→
   101→  const openEdit = (b: BrandDTO) => {
   102→    setEditing(b)
   103→    setForm(toForm(b))
   104→  }
   105→
   106→  const set = <K extends keyof BrandFormState>(key: K, value: BrandFormState[K]) =>
   107→    setForm((f) => (f ? { ...f, [key]: value } : f))
   108→
   109→  /* Dirty check — bandingkan form dengan brand asal */
   110→  const dirty = useMemo(() => {
   111→    if (!editing || !form) return false
   112→    const orig = toForm(editing)
   113→    return (Object.keys(orig) as (keyof BrandFormState)[]).some((k) => orig[k] !== form[k])
   114→  }, [editing, form])
   115→
   116→  const formErrors = useMemo(() => {
   117→    if (!form) return {} as Record<string, string>
   118→    const e: Record<string, string> = {}
   119→    if (form.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) e.email = 'Format email tidak valid'
   120→    if (form.color && !/^#[0-9a-fA-F]{6}$/.test(form.color)) e.color = 'Gunakan kode hex #rrggbb'
   121→    const sla = Number(form.slaHours)
   122→    if (!Number.isFinite(sla) || sla < 1 || sla > 168) e.slaHours = 'SLA 1–168 jam'
   123→    for (const k of ['invoicePrefix', 'quotationPrefix'] as const) {
   124→      if (!/^[A-Z]{2,6}$/.test(form[k])) e[k] = '2–6 huruf kapital'
   125→    }
   126→    return e
   127→  }, [form])
   128→
   129→  const hasErrors = Object.keys(formErrors).length > 0
   130→
   131→  const handleSave = async () => {
   132→    if (!editing || !form || !dirty || hasErrors) return
   133→    setSaving(true)
   134→    try {
   135→      const updated = await crmApi.updateBrand(editing.id, {
   136→        tagline: form.tagline,
   137→        description: form.description,
   138→        website: form.website,
   139→        email: form.email,
   140→        phone: form.phone,
   141→        instagram: form.instagram,
   142→        address: form.address,
   143→        color: form.color,
   144→        slaHours: Number(form.slaHours),
   145→        primaryCurrency: form.primaryCurrency,
   146→        invoicePrefix: form.invoicePrefix,
   147→        quotationPrefix: form.quotationPrefix,
   148→        workflowType: form.workflowType,
   149→        isActive: form.isActive,
   150→      })
   151→      setBrands(brands.map((b) => (b.id === updated.id ? updated : b)))
   152→      toast({ title: 'Pengaturan brand tersimpan', description: `${updated.name} diperbarui — kop dokumen & identifikasi kanal ikut berubah.` })
   153→      setEditing(null)
   154→      setForm(null)
   155→    } catch (e) {
   156→      toast({ title: 'Gagal menyimpan pengaturan', description: e instanceof Error ? e.message : undefined, variant: 'destructive' })
   157→    } finally { setSaving(false) }
   158→  }
   159→
   160→  const maxCount = Math.max(1, ...brands.map(b => opps.filter(o => o.executingBrandId === b.id && !['WON', 'LOST'].includes(o.stage)).length))
   161→
   162→  return (
   163→    <div className="space-y-5">
   164→      <div className="flex items-start justify-between gap-3">
   165→        <SectionHeader title="Struktur Multi-Brand" description="Satu database perusahaan — identitas, layanan, SLA, workflow, dan tim per brand" action={<RefreshButton onClick={load} loading={loading} />} />
   166→      </div>
   167→
   168→      <Card className="border-teal-100 bg-teal-50/60">
   169→        <CardContent className="flex items-start gap-3 p-4">
   170→          <Info className="mt-0.5 h-4 w-4 shrink-0 text-teal-600" />
   171→          <p className="text-xs leading-relaxed text-teal-900">
   172→            <b>Architecture note:</b> keempat brand berbagi satu database company/contact. Menambah brand kelima cukup dengan konfigurasi baru (nama, warna, layanan, SLA, workflow, template) — tanpa mengubah source code maupun menduplikasi data klien.
   173→            {canManage && <> Klik <b>Pengaturan</b> pada kartu brand untuk memperbarui identitas kontak, warna, SLA, dan prefix dokumen.</>}
   174→          </p>
   175→        </CardContent>
   176→      </Card>
   177→
   178→      {loading ? <LoadingRows rows={4} /> : (
   179→        <div className="grid gap-4 lg:grid-cols-2">
   180→          {brands.map(b => {
   181→            const brandOpps = opps.filter(o => o.executingBrandId === b.id)
   182→            const open = brandOpps.filter(o => !['WON', 'LOST'].includes(o.stage))
   183→            const openValue = open.reduce((a, o) => a + o.estimatedValue, 0)
   184→            const wonValue = brandOpps.filter(o => o.stage === 'WON').reduce((a, o) => a + o.estimatedValue, 0)
   185→            const team = users.filter(u => u.brandIds.includes(b.id))

   187→              (acc[s.category] ??= []).push(s.name); return acc
   188→            }, {})
   189→            const contacts: { icon: React.ReactNode; text: string }[] = [
   190→              ...(b.email ? [{ icon: <Mail className="h-3 w-3" />, text: b.email }] : []),
   191→              ...(b.phone ? [{ icon: <Phone className="h-3 w-3" />, text: b.phone }] : []),
   192→              ...(b.instagram ? [{ icon: <Instagram className="h-3 w-3" />, text: b.instagram }] : []),
   193→              ...(b.address ? [{ icon: <MapPin className="h-3 w-3" />, text: b.address }] : []),
   194→            ]
   195→
   196→            return (
   197→              <Card key={b.id} className="card-hover overflow-hidden rounded-xl hover:border-slate-300">
   198→                <div className="h-1.5 w-full" style={{ backgroundColor: b.color }} />
   199→                <CardHeader className="pb-2">
   200→                  <div className="flex flex-wrap items-start justify-between gap-2">
   201→                    <div className="min-w-0">
   202→                      <CardTitle className="flex items-center gap-2 text-base">
   203→                        <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg text-white shadow-sm" style={{ backgroundColor: b.color }}>
   204→                          <Building2 className="h-4 w-4" />
   205→                        </span>
   206→                        {b.name}
   207→                      </CardTitle>
   208→                      <p className="mt-1 text-xs text-slate-500">{b.tagline}</p>
   209→                    </div>
   210→                    <div className="flex shrink-0 items-center gap-1">
   211→                      <a href={b.website ? (b.website.startsWith('http') ? b.website : `https://${b.website}`) : '#'} target="_blank" rel="noreferrer" className="inline-flex min-h-[44px] items-center gap-1.5 rounded-lg px-2.5 text-[11px] font-medium text-slate-400 transition-colors hover:bg-slate-100 hover:text-teal-700 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-slate-400">
   212→                        <span className="h-2 w-2 rounded-full" style={{ backgroundColor: b.color }} aria-hidden />
   213→                        <Globe className="h-3.5 w-3.5" /> {b.website?.replace(/^https?:\/\//, '')}
   214→                      </a>
   215→                      {canManage && (
   216→                        <Button
   217→                          variant="outline"
   218→                          size="sm"
   219→                          onClick={() => openEdit(b)}
   220→                          className="h-8 gap-1.5 px-2.5 text-xs"
   221→                          aria-label={`Pengaturan ${b.name}`}
   222→                        >
   223→                          <Settings2 className="h-3.5 w-3.5" /> Pengaturan
   224→                        </Button>
   225→                      )}
   226→                    </div>
   227→                  </div>
   228→                </CardHeader>
   229→                <CardContent className="space-y-4">
   230→                  <div className="flex flex-wrap gap-1.5">
   231→                    <Badge variant="secondary" className="gap-1 border-0 bg-slate-100 text-[10px] text-slate-600"><Timer className="h-3 w-3" /> SLA {b.slaHours} jam</Badge>
   232→                    <Badge variant="secondary" className="gap-1 border-0 bg-slate-100 text-[10px] tabular-nums text-slate-600">💰 {b.primaryCurrency}</Badge>
   233→                    <Badge variant="secondary" className="gap-1 border-0 bg-slate-100 text-[10px] tabular-nums text-slate-600">Invoice: {b.invoicePrefix}-YYYY-####</Badge>
   234→                    <Badge variant="secondary" className="gap-1 border-0 bg-slate-100 text-[10px] tabular-nums text-slate-600">Quotation: {b.quotationPrefix}-YYYY-####</Badge>
   235→                    <Badge variant="secondary" className="gap-1 border-0 bg-slate-100 text-[10px] text-slate-600"><Workflow className="h-3 w-3" /> {WORKFLOW_LABELS[b.workflowType] ?? b.workflowType}</Badge>
   236→                  </div>
   237→
   238→                  {contacts.length > 0 && (
   239→                    <div className="grid gap-x-4 gap-y-1.5 rounded-lg border border-slate-100 bg-slate-50/70 p-2.5 sm:grid-cols-2">
   240→                      {contacts.map((c, i) => (
   241→                        <span key={i} className="flex min-w-0 items-center gap-1.5 text-[11px] text-slate-600">
   242→                          <span className="shrink-0 text-slate-400">{c.icon}</span>
   243→                          <span className="truncate" title={c.text}>{c.text}</span>
   244→                        </span>
   245→                      ))}
   246→                    </div>
   247→                  )}
   248→
   249→                  <div>
   250→                    <div className="mb-1 flex items-center justify-between text-[11px]">
   251→                      <span className="font-medium text-slate-600">Pipeline aktif · {open.length} opportunity</span>
   252→                      <span className="font-bold tabular-nums text-slate-800">{formatMoney(openValue, 'IDR', true)}</span>
   253→                    </div>
   254→                    {/* Warna bar mengikuti brand via CSS var — indicator Progress memakai bg-[var(--brand-c)] */}
   255→                    <Progress value={(open.length / maxCount) * 100} className="h-2 [&>div]:bg-[var(--brand-c)]" style={{ ['--brand-c' as string]: b.color }} />
   256→                    <p className="mt-1 text-[11px] text-slate-500">Total Won: <span className="font-semibold tabular-nums text-emerald-700">{formatMoney(wonValue, 'IDR', true)}</span></p>
   257→                  </div>
   258→
   259→                  <div>
   260→                    <p className="mb-1.5 text-[11px] font-semibold uppercase tracking-wider text-slate-400">Layanan ({b.services.length})</p>
   261→                    <div className="space-y-2">
   262→                      {Object.entries(byCategory).map(([cat, list]) => (
   263→                        <div key={cat}>
   264→                          <p className="text-[11px] font-medium text-slate-500">{cat}</p>
   265→                          <div className="mt-1 flex flex-wrap gap-1">
   266→                            {list.map(s => (
   267→                              <span key={s} className="rounded-md px-1.5 py-0.5 text-[10px] font-medium" style={{ backgroundColor: `${b.color}10`, color: b.color }}>{s}</span>
   268→                            ))}
   269→                          </div>
   270→                        </div>
   271→                      ))}
   272→                    </div>
   273→                  </div>
   274→
   275→                  <div>
   276→                    <p className="mb-1.5 flex items-center gap-1 text-[11px] font-semibold uppercase tracking-wider text-slate-400"><Users2 className="h-3 w-3" /> Tim ({team.length})</p>
   277→                    <div className="flex flex-wrap gap-1.5">
   278→                      {team.map(u => (
   279→                        <span key={u.id} className="inline-flex items-center gap-1.5 rounded-full border border-slate-200 bg-white py-0.5 pl-0.5 pr-2 text-[11px] text-slate-600">
   280→                          <UserAvatar name={u.name} color={u.avatarColor} size={18} /> {u.name}
   281→                        </span>
   282→                      ))}
   283→                    </div>
   284→                  </div>
   285→
   286→                  <details className="rounded-lg bg-slate-50 p-2.5">
   287→                    <summary className="flex min-h-[44px] cursor-pointer select-none items-center gap-1 rounded text-[11px] font-medium text-slate-600 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-slate-400">Workflow produksi standar</summary>
   288→                    <p className="mt-1.5 text-[11px] text-slate-500">{(WORKFLOW_MILESTONES[b.workflowType] ?? []).join(' → ')}</p>
   289→                  </details>
   290→                </CardContent>
   291→              </Card>
   292→            )
   293→          })}
   294→        </div>
   295→      )}
   296→
   297→      {/* ===== Dialog Pengaturan Brand (live preview kop dokumen) ===== */}
   298→      <Dialog open={!!editing && !!form} onOpenChange={(v) => { if (!v) { setEditing(null); setForm(null) } }}>
   299→        <DialogContent className="max-h-[92vh] overflow-y-auto sm:max-w-2xl">
   300→          <DialogHeader>
   301→            <DialogTitle className="flex items-center gap-2">
   302→              <span className="flex h-7 w-7 items-center justify-center rounded-md text-white" style={{ backgroundColor: editing?.color }}>
   303→                <Building2 className="h-3.5 w-3.5" />
   304→              </span>
   305→              Pengaturan · {editing?.name}
   306→            </DialogTitle>
   307→            <DialogDescription>
   308→              Identitas kontak dipakai di kop dokumen (penawaran/invoice) & membantu identifikasi kanal masuk. Warna dipakai konsisten di seluruh aplikasi.
   309→            </DialogDescription>
   310→          </DialogHeader>
   311→
   312→          {form && editing && (
   313→            <div className="grid gap-5 md:grid-cols-[1fr_240px]">
   314→              {/* Kolom form */}
   315→              <div className="space-y-4">
   316→                <div className="space-y-3">
   317→                  <p className="text-[11px] font-semibold uppercase tracking-wider text-slate-400">Identitas</p>
   318→                  <div className="grid gap-3 sm:grid-cols-2">
   319→                    <div className="space-y-1.5">
   320→                      <Label htmlFor="brand-tagline" className="text-xs">Tagline</Label>
   321→                      <Input id="brand-tagline" value={form.tagline} onChange={(e) => set('tagline', e.target.value)} placeholder="mis. Video production house" className="h-9" />
   322→                    </div>
   323→                    <div className="space-y-1.5">
   324→                      <Label htmlFor="brand-workflow" className="text-xs">Workflow produksi</Label>
   325→                      <Select value={form.workflowType} onValueChange={(v) => set('workflowType', v)}>
   326→                        <SelectTrigger id="brand-workflow" className="h-9"><SelectValue /></SelectTrigger>
   327→                        <SelectContent>
   328→                          {WORKFLOWS.map((w) => <SelectItem key={w} value={w}>{WORKFLOW_LABELS[w]}</SelectItem>)}
   329→                        </SelectContent>
   330→                      </Select>
   331→                    </div>
   332→                  </div>
   333→                  <div className="space-y-1.5">
   334→                    <Label htmlFor="brand-desc" className="text-xs">Deskripsi</Label>
   335→                    <Textarea id="brand-desc" value={form.description} onChange={(e) => set('description', e.target.value)} rows={2} placeholder="Deskripsi singkat brand untuk konteks tim" className="min-h-0 resize-none py-2" />
   336→                  </div>
   337→                </div>
   338→
   339→                <div className="space-y-3">
   340→                  <p className="text-[11px] font-semibold uppercase tracking-wider text-slate-400">Kontak & dokumen</p>
   341→                  <div className="grid gap-3 sm:grid-cols-2">
   342→                    <div className="space-y-1.5">
   343→                      <Label htmlFor="brand-email" className="text-xs">Email</Label>
   344→                      <Input id="brand-email" type="email" value={form.email} onChange={(e) => set('email', e.target.value)} placeholder="hello@brand.co.id" className="h-9" aria-invalid={!!formErrors.email} />
   345→                      {formErrors.email && <p className="text-[10px] text-rose-600">{formErrors.email}</p>}
   346→                    </div>
   347→                    <div className="space-y-1.5">
   348→                      <Label htmlFor="brand-phone" className="text-xs">Telepon / WA bisnis</Label>
   349→                      <Input id="brand-phone" value={form.phone} onChange={(e) => set('phone', e.target.value)} placeholder="+62 812 3456 7890" className="h-9" />
   350→                    </div>
   351→                    <div className="space-y-1.5">
   352→                      <Label htmlFor="brand-ig" className="text-xs">Instagram</Label>
   353→                      <Input id="brand-ig" value={form.instagram} onChange={(e) => set('instagram', e.target.value)} placeholder="@brand.official" className="h-9" />
   354→                    </div>
   355→                    <div className="space-y-1.5">
   356→                      <Label htmlFor="brand-website" className="text-xs">Website</Label>
   357→                      <Input id="brand-website" value={form.website} onChange={(e) => set('website', e.target.value)} placeholder="www.brand.co.id" className="h-9" />
   358→                    </div>
   359→                  </div>
   360→                  <div className="space-y-1.5">
   361→                    <Label htmlFor="brand-address" className="text-xs">Alamat (untuk kop dokumen)</Label>
   362→                    <Textarea id="brand-address" value={form.address} onChange={(e) => set('address', e.target.value)} rows={2} placeholder="Jl. … No. …, Kota" className="min-h-0 resize-none py-2" />
   363→                  </div>
   364→                </div>
   365→
   366→                <div className="space-y-3">
   367→                  <p className="text-[11px] font-semibold uppercase tracking-wider text-slate-400">Konfigurasi</p>

   369→                    <div className="space-y-1.5">
   370→                      <Label htmlFor="brand-color" className="text-xs">Warna brand</Label>
   371→                      <div className="flex items-center gap-2">
   372→                        <input
   373→                          id="brand-color"

   375→                          value={/^#[0-9a-fA-F]{6}$/.test(form.color) ? form.color : '#10b981'}
   376→                          onChange={(e) => set('color', e.target.value)}
   377→                          className="h-9 w-9 shrink-0 cursor-pointer rounded-md border border-slate-200 bg-white p-0.5"
   378→                          aria-label="Pilih warna brand"
   379→                        />

   383→                        {COLOR_PRESETS.map((c) => (
