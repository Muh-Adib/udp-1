    72→
    73→export default function InboxView() {
    74→  const brands = useCrmStore((s) => s.brands)
    75→  const openOpportunity = useCrmStore((s) => s.openOpportunity)
    76→  const { toast } = useToast()
    77→
    78→  /* ---------- Filter state ---------- */
    79→  const [searchInput, setSearchInput] = useState('')
    80→  const [search, setSearch] = useState('')
    81→  const [brandId, setBrandId] = useState('')
    82→  const [channel, setChannel] = useState('')
    83→  const [unreadOnly, setUnreadOnly] = useState(false)
    84→
    85→  /* ---------- Data state ---------- */
    86→  const [interactions, setInteractions] = useState<InteractionDTO[]>([])
    87→  const [opportunities, setOpportunities] = useState<OpportunityDTO[]>([])
    88→  const [loading, setLoading] = useState(true)
    89→  const [error, setError] = useState<string | null>(null)
    90→
    91→  /* ---------- Duplicate state ---------- */
    92→  const [duplicates, setDuplicates] = useState<DuplicateCandidate[]>([])
    93→  const [dismissed, setDismissed] = useState<Set<string>>(new Set())
    94→  const [dupOpen, setDupOpen] = useState(false)
    95→  const [mergingKey, setMergingKey] = useState<string | null>(null)
    96→
    97→  /* ---------- Dialog state ---------- */
    98→  const [replyTarget, setReplyTarget] = useState<InteractionDTO | null>(null)
    99→  const [replyChannel, setReplyChannel] = useState('WHATSAPP')
   100→  const [replyBody, setReplyBody] = useState('')
   101→  const [sending, setSending] = useState(false)
   102→
   103→  const [createTarget, setCreateTarget] = useState<InteractionDTO | null>(null)
   104→  const [oppForm, setOppForm] = useState({ title: '', brandId: '', serviceId: NONE, estimatedValue: '', brief: '', priority: 'MEDIUM' })
   105→  const [creating, setCreating] = useState(false)
   106→
   107→  const [webFormOpen, setWebFormOpen] = useState(false)
   108→  const [wf, setWf] = useState({ name: '', email: '', whatsapp: '', brandId: '', serviceId: NONE, message: '' })
   109→  const [submittingWf, setSubmittingWf] = useState(false)
   110→
   111→  /* ---------- Debounce search 400ms ---------- */
   112→  useEffect(() => {
   113→    const t = setTimeout(() => setSearch(searchInput.trim()), 400)
   114→    return () => clearTimeout(t)
   115→  }, [searchInput])
   116→
   117→  /* ---------- Query params (hanya yang terisi) ---------- */
   118→  const params = useMemo(() => {
   119→    const p = new URLSearchParams()
   120→    if (brandId) p.set('brandId', brandId)
   121→    if (channel) p.set('channel', channel)
   122→    if (unreadOnly) p.set('unreadOnly', '1')
   123→    if (search) p.set('search', search)
   124→    return p.toString()
   125→  }, [brandId, channel, unreadOnly, search])
   126→
   127→  const loadInteractions = useCallback(async () => {
   128→    setLoading(true)
   129→    setError(null)
   130→    try {
   131→      // opportunities ikut dimuat untuk komputasi SLA per lead (stage/createdAt/brand)
   132→      const [msgs, opps] = await Promise.all([crmApi.interactions(params), crmApi.opportunities()])
   133→      setInteractions(msgs)
   134→      setOpportunities(opps)
   135→    } catch (e) {
   136→      setError(e instanceof Error ? e.message : 'Gagal memuat pesan')
   137→    } finally {
   138→      setLoading(false)
   139→    }
   140→  }, [params])
   141→
   142→  const loadDuplicates = useCallback(async () => {
   143→    try {
   144→      setDuplicates(await crmApi.duplicates())
   145→    } catch {
   146→      /* banner duplikat bersifat opsional — gagal senyap */
   147→    }
   148→  }, [])
   149→
   150→  useEffect(() => { void loadInteractions() }, [loadInteractions])
   151→  useEffect(() => { void loadDuplicates() }, [loadDuplicates])
   152→
   153→  const visibleDuplicates = useMemo(
   154→    () => duplicates.filter((d) => !dismissed.has(dupKey(d))),
   155→    [duplicates, dismissed],
   156→  )
   157→  const unreadCount = interactions.filter((m) => !m.replied).length
   158→
   159→  /* ---------- SLA breach per lead (client-side, sama dgn dashboard) ---------- */
   160→  // waitingSince = lastInboundAt (interaksi IN terakhir, fallback createdAt utk lead
   161→  // yang belum pernah ada inbound). Breach bila waitingHours > slaHours brand
   162→  // (sourceBrand ?? executingBrand, fallback 24).
   163→  const slaBreaches = useMemo(() => {
   164→    const map = new Map<string, { over: number; sla: number; brandName: string }>()
   165→    const nowMs = Date.now()
   166→    for (const o of opportunities) {
   167→      if (o.stage !== 'NEW' && o.stage !== 'CONTACT_ATTEMPTED') continue
   168→      const brand = brands.find((b) => b.id === o.sourceBrandId) ?? brands.find((b) => b.id === o.executingBrandId)
   169→      const sla = brand?.slaHours ?? 24
   170→      const waitingHours = (nowMs - new Date(o.lastInboundAt ?? o.createdAt).getTime()) / 3600000
   171→      if (waitingHours > sla) {
   172→        map.set(o.id, { over: Math.round(waitingHours - sla), sla, brandName: brand?.name ?? o.brandName })
   173→      }
   174→    }
   175→    return map
   176→  }, [opportunities, brands])
   177→
   178→  /* ---------- Aksi: reply ---------- */
   179→  const openReply = (m: InteractionDTO) => {
   180→    setReplyChannel(m.channel === 'EMAIL' ? 'EMAIL' : 'WHATSAPP')
   181→    setReplyBody('')
   182→    setReplyTarget(m)
   183→  }
   184→
   185→  const handleSendReply = async () => {
   186→    if (!replyTarget || !replyBody.trim()) return
   187→    setSending(true)
   188→    try {
   189→      const body = replyBody.trim()
   190→      if (replyTarget.opportunityId) {
   191→        await crmApi.addInteraction(replyTarget.opportunityId, { channel: replyChannel, direction: 'OUT', body })
   192→      } else {
   193→        await crmApi.logInteraction({
   194→          contactId: replyTarget.contactId,
   195→          brandId: replyTarget.brandId,
   196→          channel: replyChannel,
   197→          direction: 'OUT',
   198→          body,
   199→        })
   200→      }
   201→      toast({ title: 'Balasan terkirim — draft disimpan di percakapan', description: `Kepada ${replyTarget.contactName} via ${channelMeta(replyChannel).label}.` })
   202→      setReplyTarget(null)
   203→      setReplyBody('')
   204→      await loadInteractions()
   205→    } catch (e) {
   206→      toast({
   207→        title: 'Gagal mengirim balasan',
   208→        description: e instanceof Error ? e.message : 'Coba lagi',
   209→        variant: 'destructive',
   210→      })
   211→    } finally {
   212→      setSending(false)
   213→    }
   214→  }
   215→
   216→  /* ---------- Aksi: create opportunity ---------- */
   217→  useEffect(() => {
   218→    if (createTarget) {
   219→      setOppForm({
   220→        title: `Kebutuhan ${createTarget.brandName}`,
   221→        brandId: createTarget.brandId,
   222→        serviceId: NONE,
   223→        estimatedValue: '',
   224→        brief: createTarget.body,
   225→        priority: 'MEDIUM',
   226→      })
   227→    }
   228→  }, [createTarget])
   229→
   230→  const selectedBrand = brands.find((b) => b.id === oppForm.brandId)
   231→  const brandServices = selectedBrand?.services ?? []
   232→
   233→  /** API opportunity butuh companyId — resolve dari kontak via endpoint contacts. */
   234→  const resolveCompanyId = async (contactId: string, contactName: string): Promise<string | null> => {
   235→    try {
   236→      const list = await crmApi.contacts(`search=${encodeURIComponent(contactName)}`)
   237→      const hit = list.find((c) => c.id === contactId)
   238→      if (hit?.companyId) return hit.companyId
   239→      const all = await crmApi.contacts()
   240→      return all.find((c) => c.id === contactId)?.companyId ?? null
   241→    } catch {
   242→      return null
   243→    }
   244→  }
   245→
   246→  const handleCreateOpp = async () => {
   247→    if (!createTarget) return
   248→    if (!oppForm.title.trim()) {
   249→      toast({ title: 'Judul wajib diisi', variant: 'destructive' })
   250→      return
   251→    }
   252→    if (!oppForm.brandId) {
   253→      toast({ title: 'Brand wajib dipilih', variant: 'destructive' })
   254→      return
   255→    }
   256→    setCreating(true)
   257→    try {
   258→      const companyId = await resolveCompanyId(createTarget.contactId, createTarget.contactName)
   259→      if (!companyId) {
   260→        throw new Error('Company untuk kontak ini tidak ditemukan. Buat company terlebih dahulu di modul Contacts.')
   261→      }
   262→      const opp = await crmApi.createOpportunity({
   263→        title: oppForm.title.trim(),
   264→        companyId,
   265→        contactId: createTarget.contactId,
   266→        sourceBrandId: oppForm.brandId,
   267→        executingBrandId: oppForm.brandId,
   268→        serviceId: oppForm.serviceId === NONE ? undefined : oppForm.serviceId,
   269→        leadSource: CHANNEL_TO_LEAD_SOURCE[createTarget.channel] ?? 'WEBSITE',
   270→        channel: createTarget.channel,
   271→        estimatedValue: oppForm.estimatedValue ? Number(oppForm.estimatedValue) : 0,
   272→        brief: oppForm.brief.trim() || undefined,
   273→        priority: oppForm.priority,
   274→      })
   275→      toast({ title: 'Opportunity dibuat', description: `${opp.code} — ${opp.title}` })
   276→      setCreateTarget(null)
   277→      openOpportunity(opp.id)
   278→    } catch (e) {
   279→      toast({
   280→        title: 'Gagal membuat opportunity',
   281→        description: e instanceof Error ? e.message : 'Coba lagi',
   282→        variant: 'destructive',
   283→      })
   284→    } finally {
   285→      setCreating(false)
   286→    }
   287→  }
   288→
   289→  /* ---------- Aksi: merge / keep separate ---------- */
   290→  const handleMerge = async (c: DuplicateCandidate) => {
   291→    const key = dupKey(c)
   292→    setMergingKey(key)
   293→    try {
   294→      await crmApi.merge({ keepId: c.contactA.id, mergeId: c.contactB.id })
   295→      toast({
   296→        title: 'Duplikat digabung',
   297→        description: `${c.contactA.fullName} dipertahankan; data ${c.contactB.fullName} digabungkan ke kontak utama.`,
   298→      })
   299→      setDupOpen(false)
   300→      await Promise.all([loadDuplicates(), loadInteractions()])
   301→    } catch (e) {
   302→      toast({
   303→        title: 'Gagal menggabungkan duplikat',
   304→        description: e instanceof Error ? e.message : 'Coba lagi',
   305→        variant: 'destructive',
   306→      })
   307→    } finally {
   308→      setMergingKey(null)
   309→    }
   310→  }
   311→
   312→  const handleKeepSeparate = (c: DuplicateCandidate) => {
   313→    setDismissed((prev) => new Set(prev).add(dupKey(c)))
   314→    toast({ title: 'Kandidat diabaikan', description: `${c.contactA.fullName} & ${c.contactB.fullName} dibiarkan terpisah.` })
   315→  }
   316→
   317→  /* ---------- Aksi: simulasi form website ---------- */
   318→  const handleWebsiteForm = async () => {
   319→    if (!wf.name.trim() || !wf.email.trim() || !wf.brandId || !wf.message.trim()) {
   320→      toast({ title: 'Lengkapi nama, email, brand, dan pesan', variant: 'destructive' })
   321→      return
   322→    }
   323→    if (!/.+@.+\..+/.test(wf.email.trim())) {
   324→      toast({ title: 'Format email tidak valid', variant: 'destructive' })
   325→      return
   326→    }
   327→    setSubmittingWf(true)
   328→    try {
   329→      await crmApi.websiteForm({
   330→        websiteForm: {
   331→          name: wf.name.trim(),
   332→          email: wf.email.trim(),
   333→          whatsapp: wf.whatsapp.trim() || undefined,
   334→          message: wf.message.trim(),
   335→          brandId: wf.brandId,
   336→          serviceId: wf.serviceId === NONE ? undefined : wf.serviceId,
   337→        },
   338→      })
   339→      toast({ title: 'Lead baru dari website masuk ke inbox', description: `${wf.name.trim()} — ${brands.find((b) => b.id === wf.brandId)?.name ?? ''}` })
   340→      setWebFormOpen(false)
   341→      setWf({ name: '', email: '', whatsapp: '', brandId: '', serviceId: NONE, message: '' })
   342→      await loadInteractions()
   343→    } catch (e) {
   344→      toast({
   345→        title: 'Gagal mensimulasikan lead website',
   346→        description: e instanceof Error ? e.message : 'Coba lagi',
   347→        variant: 'destructive',
   348→      })
   349→    } finally {
   350→      setSubmittingWf(false)
   351→    }
   352→  }
   353→
   354→  const wfServices = brands.find((b) => b.id === wf.brandId)?.services ?? []
   355→
   356→  /* ---------- Render ---------- */
   357→  return (
   358→    <div className="space-y-4">
   359→      {/* ===== Toolbar ===== */}
   360→      <Card className="rounded-xl border-slate-200 shadow-sm">
   361→        <CardContent className="flex flex-wrap items-center gap-2 p-3 sm:p-4">
   362→          <div className="relative min-w-[200px] flex-1">
   363→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
   364→            <Input
   365→              id="inbox-search"
   366→              value={searchInput}
   367→              onChange={(e) => setSearchInput(e.target.value)}
   368→              onKeyDown={(e) => { if (e.key === 'Enter') setSearch(searchInput.trim()) }}
   369→              placeholder="Cari nama, isi pesan…"
   370→              className="h-9 pl-9"
   371→              aria-label="Cari pesan"
   372→            />
   373→          </div>
   374→
   375→          <Select value={brandId || ALL} onValueChange={(v) => setBrandId(v === ALL ? '' : v)}>
   376→            <SelectTrigger className="h-9 w-[150px] sm:w-[170px]" aria-label="Filter brand">
   377→              <SelectValue placeholder="Semua Brand" />
   378→            </SelectTrigger>
   379→            <SelectContent>
   380→              <SelectItem value={ALL}>Semua Brand</SelectItem>
   381→              {brands.map((b) => (
   382→                <SelectItem key={b.id} value={b.id}>{b.name}</SelectItem>
   383→              ))}
   384→            </SelectContent>
   385→          </Select>
   386→
   387→          <Select value={channel || ALL} onValueChange={(v) => setChannel(v === ALL ? '' : v)}>
   388→            <SelectTrigger className="h-9 w-[140px] sm:w-[150px]" aria-label="Filter kanal">
   389→              <SelectValue placeholder="Semua Kanal" />
   390→            </SelectTrigger>
   391→            <SelectContent>
   392→              <SelectItem value={ALL}>Semua Kanal</SelectItem>
   393→              {CHANNELS.map((c) => (
   394→                <SelectItem key={c.key} value={c.key}>{c.label}</SelectItem>
   395→              ))}
   396→            </SelectContent>
   397→          </Select>
   398→
   399→          <label htmlFor="inbox-unread" className="flex cursor-pointer items-center gap-2 rounded-lg px-2 py-2 text-xs font-medium text-slate-600 hover:bg-slate-50">
   400→            <Switch id="inbox-unread" checked={unreadOnly} onCheckedChange={setUnreadOnly} />
   401→            Belum dibalas saja
   402→          </label>
   403→
   404→          <div className="ml-auto flex items-center gap-2">
   405→            <RefreshButton onClick={() => void loadInteractions()} loading={loading} />
   406→            <Button variant="outline" size="sm" className="h-8 gap-1.5" onClick={() => setWebFormOpen(true)}>
   407→              <Globe className="h-3.5 w-3.5" />
   408→              <span className="hidden sm:inline">Simulasi Form Website</span>
   409→              <span className="sm:hidden">Form Website</span>
   410→            </Button>
   411→          </div>
   412→        </CardContent>
   413→      </Card>
   414→
   415→      {/* ===== Banner duplikat ===== */}
   416→      {visibleDuplicates.length > 0 && (
   417→        <Alert className="rounded-xl border-amber-300 bg-amber-50 text-amber-900">
   418→          <AlertTriangle className="h-4 w-4 text-amber-600" />
   419→          <AlertTitle className="text-sm font-semibold">⚠️ {visibleDuplicates.length} kandidat duplikat terdeteksi</AlertTitle>
   420→          <AlertDescription className="flex flex-wrap items-center justify-between gap-3">
   421→            <span className="text-amber-800">
   422→              Kontak ganda memecah timeline percakapan dan merusak statistik pipeline. Review dan gabungkan sekarang.
   423→            </span>
   424→            <Button size="sm" className="h-8 gap-1.5 bg-amber-600 text-white hover:bg-amber-700" onClick={() => setDupOpen(true)}>
   425→              <Merge className="h-3.5 w-3.5" /> Review Duplikat
   426→            </Button>
   427→          </AlertDescription>
   428→        </Alert>
   429→      )}
   430→
   431→      {/* ===== Daftar pesan ===== */}
   432→      <Card className="rounded-xl border-slate-200 shadow-sm">
   433→        <CardHeader className="pb-3">
   434→          <CardTitle className="flex items-center gap-2 text-base">
   435→            <Inbox className="h-4 w-4 text-teal-600" /> Pesan Masuk
   436→          </CardTitle>
   437→          <CardDescription>
   438→            {loading ? 'Memuat…' : `${interactions.length} pesan · ${unreadCount} belum dibalas`}
   439→          </CardDescription>
   440→        </CardHeader>
   441→        <CardContent className="p-3 sm:p-4 sm:pt-0">
   442→          {error ? (
   443→            <div className="flex flex-col items-center gap-3 rounded-xl border border-rose-200 bg-rose-50 p-6 text-center">
   444→              <AlertTriangle className="h-7 w-7 text-rose-500" />
   445→              <div>
   446→                <p className="font-semibold text-rose-800">Gagal memuat pesan</p>
   447→                <p className="mt-0.5 text-sm text-rose-600">{error}</p>
   448→              </div>
   449→              <Button variant="outline" className="border-rose-300 hover:bg-rose-100" onClick={() => void loadInteractions()}>
   450→                Coba lagi
   451→              </Button>
   452→            </div>
   453→          ) : loading && interactions.length === 0 ? (
   454→            <LoadingRows rows={6} />
   455→          ) : interactions.length === 0 ? (
   456→            <EmptyState
   457→              icon={<Inbox className="h-6 w-6" />}
   458→              title="Tidak ada pesan masuk"
   459→              description="Coba ubah filter, matikan “Belum dibalas saja”, atau simulasikan lead masuk dari form website."
   460→              action={
   461→                <Button variant="outline" size="sm" className="gap-1.5" onClick={() => setWebFormOpen(true)}>
   462→                  <Globe className="h-3.5 w-3.5" /> Simulasi Form Website
   463→                </Button>
   464→              }
   465→            />
   466→          ) : (
   467→            <div className={cn('max-h-[65vh] space-y-2 overflow-y-auto pr-1', SCROLLBAR)}>
