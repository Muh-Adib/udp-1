   103→/* ---------------- Invoice row ---------------- */
   104→function InvoiceRow({ inv, onPay, onDetail }: { inv: InvoiceDTO; onPay: (inv: InvoiceDTO) => void; onDetail: (inv: InvoiceDTO) => void }) {
   105→  const overdue = isOverdue(inv)
   106→  const pct = inv.total > 0 ? Math.min(100, Math.max(0, (inv.paidAmount / inv.total) * 100)) : 0
   107→  const canPay = inv.status !== 'PAID' && inv.status !== 'CANCELLED'
   108→  return (
   109→    <div className="flex flex-col gap-3 p-4 transition-colors hover:bg-slate-50/70 md:flex-row md:items-center md:gap-5">
   110→      {/* Kiri: identitas */}
   111→      <div className="min-w-0 flex-1">
   112→        <div className="flex flex-wrap items-center gap-2">
   113→          <span className="font-mono text-[11px] text-slate-400">{inv.code}</span>
   114→          <BrandChip name={inv.brandName} color={inv.brandColor} size="xs" />
   115→          {overdue && <span className="rounded-full bg-rose-100 px-2 py-0.5 text-[10px] font-semibold text-rose-700">Lewat jatuh tempo</span>}
   116→        </div>
   117→        <p className="mt-0.5 truncate text-sm font-semibold text-slate-900">{inv.title}</p>
   118→        <div className="mt-0.5 flex flex-wrap items-center gap-x-3 gap-y-0.5 text-[11px] text-slate-500">
   119→          <span className="inline-flex items-center gap-1"><Building2 className="h-3 w-3" />{inv.companyName}</span>
   120→          {inv.dueDate && (
   121→            <span className={cn('inline-flex items-center gap-1', overdue ? 'font-semibold text-rose-600' : 'text-slate-500')}>
   122→              <CalendarClock className="h-3 w-3" />
   123→              Jatuh tempo {formatDate(inv.dueDate)}
   124→            </span>
   125→          )}
   126→        </div>
   127→      </div>
   128→
   129→      {/* Tengah: progres pembayaran */}
   130→      <div className="w-full shrink-0 md:w-60">
   131→        <div className="flex items-center justify-between gap-2 text-[11px]">
   132→          <span className="truncate text-slate-500 tabular-nums">
   133→            Terbayar <span className="font-semibold text-slate-700">{formatMoney(inv.paidAmount, inv.currency, true)}</span>
   134→            {' '}/ {formatMoney(inv.total, inv.currency, true)}
   135→          </span>
   136→          <Badge variant="outline" className="h-4 shrink-0 px-1.5 text-[10px] font-medium text-slate-500">{inv.payments.length}× bayar</Badge>
   137→        </div>
   138→        <Progress value={pct} className="mt-1.5 h-1.5 bg-slate-100 [&>div]:bg-emerald-500" />
   139→      </div>
   140→
   141→      {/* Kanan: status + aksi */}
   142→      <div className="flex items-center gap-2 md:justify-end">
   143→        <StatusBadge status={inv.status} />
   144→        <Button
   145→          size="sm"
   146→          className="h-8 gap-1.5 bg-teal-600 text-white hover:bg-teal-700"
   147→          disabled={!canPay}
   148→          title={canPay ? 'Catat pembayaran untuk invoice ini' : 'Invoice lunas / dibatalkan'}
   149→          onClick={() => onPay(inv)}
   150→        >
   151→          <Wallet className="h-3.5 w-3.5" />
   152→          <span className="hidden sm:inline">Catat Pembayaran</span>
   153→          <span className="sm:hidden">Bayar</span>
   154→        </Button>
   155→        <Button
   156→          size="icon"
   157→          variant="ghost"
   158→          className="h-8 w-8 text-slate-400 hover:text-slate-700"
   159→          aria-label={`Lihat detail invoice ${inv.code}`}
   160→          title="Lihat detail"
   161→          onClick={() => onDetail(inv)}
   162→        >
   163→          <Eye className="h-4 w-4" />
   164→        </Button>
   165→      </div>
   166→    </div>
   167→  )
   168→}

   280→    }

   313→      refreshSummary()
   314→      toast({ title: `Invoice ${updated.code} dibatalkan`, description: 'Status invoice kini Dibatalkan.' })

   316→    } catch (e) {

   319→      setCancelSubmitting(false)

   326→    setFQuoteId('none')
   327→    setFTitle('')
   328→    setFAmount('')
   329→    setFTaxPct('0')

   347→        title: 'Gagal memuat data deal / quotation',
   348→        description: e instanceof Error ? e.message : undefined,
   349→        variant: 'destructive',
   350→      }))
   351→      .finally(() => setOptsLoading(false))

   356→    setCreateError(null)
   357→    const opp = wonOpps.find(o => o.id === v)

   362→  }

   383→      const created = await financeApi.createInvoice({
   384→        opportunityId: fOppId,
   385→        quotationId: fQuoteId !== 'none' ? fQuoteId : undefined,

   412→  }, [summary])
   413→  const agingMax = Math.max(...agingRows.map(r => r.value), 1)
   414→  const agingAllZero = agingRows.length > 0 && agingRows.every(r => r.count === 0)
   415→  const brandMax = summary ? Math.max(...summary.byBrand.map(b => b.invoiced), 1) : 1
   416→  const paySisa = payInv ? payInv.total - payInv.paidAmount : 0

   418→  /* ================================================================ */

   421→      <SectionHeader
   422→        title="Finance"
   423→        description="Invoice, pembayaran, dan analitik piutang lintas brand"
