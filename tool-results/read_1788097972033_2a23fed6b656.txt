   112→function fmtRp(n: number): string {

   116→function fmtDate(iso?: string | null): string {
   117→  return iso ? dateFmt.format(new Date(iso)) : "—";

   120→function formatBytes(bytes: number): string {
   121→  if (bytes >= 1024 * 1024) return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   122→  if (bytes >= 1024) return `${Math.round(bytes / 1024)} KB`;
   123→  return `${bytes} B`;
   124→}
   125→
   126→/** Tanggal relatif id-ID (butuh `now` dari state agar purity-safe saat render). */
   127→function timeAgoId(iso: string, now: number): string {
   128→  if (!now) return fmtDate(iso);
   129→  const diff = now - new Date(iso).getTime();
   130→  if (diff < 60_000) return "baru saja";
   131→  const m = Math.floor(diff / 60_000);
   132→  if (m < 60) return `${m} menit lalu`;
   133→  const h = Math.floor(m / 60);
   134→  if (h < 24) return `${h} jam lalu`;
   135→  const d = Math.floor(h / 24);
   136→  if (d < 30) return `${d} hari lalu`;
   137→  return fmtDate(iso);
   138→}

   140→function groupDeliverables(list: DeliverableWithProject[], projects: ProjectDTO[]): DeliverableGroup[] {
   141→  const projById = new Map(projects.map((p) => [p.id, p]));
   142→  const groups: DeliverableGroup[] = [];
   143→  for (const d of list) {
   144→    let g = groups.find((x) => x.projectId === d.projectId);
   145→    if (!g) {
   146→      const proj = projById.get(d.projectId);
   147→      g = {
   148→        projectId: d.projectId,
   149→        code: d.projectCode,
   150→        name: d.projectName,
   151→        status: proj?.status ?? null,
   152→        items: [],
   153→      };
   154→      groups.push(g);
   155→    }
   156→    g.items.push(d);
   157→  }
   158→  return groups;
   159→}
