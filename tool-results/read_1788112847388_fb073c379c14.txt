   111→const rpFmt = new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", maximumFractionDigits: 0 });
   112→const dateFmt = new Intl.DateTimeFormat("id-ID", { day: "numeric", month: "short", year: "numeric" });

   114→function fmtRp(n: number): string {
   115→  return rpFmt.format(n);
   116→}
   117→
   118→function fmtDate(iso?: string | null): string {
   119→  return iso ? dateFmt.format(new Date(iso)) : "—";
   120→}
   121→
   122→function formatBytes(bytes: number): string {
   123→  if (bytes >= 1024 * 1024) return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   124→  if (bytes >= 1024) return `${Math.round(bytes / 1024)} KB`;
   125→  return `${bytes} B`;
   126→}
   127→
   128→/** Tanggal relatif id-ID (butuh `now` dari state agar purity-safe saat render). */
   129→function timeAgoId(iso: string, now: number): string {
   130→  if (!now) return fmtDate(iso);
   131→  const diff = now - new Date(iso).getTime();
   132→  if (diff < 60_000) return "baru saja";
   133→  const m = Math.floor(diff / 60_000);
   134→  if (m < 60) return `${m} menit lalu`;
   135→  const h = Math.floor(m / 60);
   136→  if (h < 24) return `${h} jam lalu`;
   137→  const d = Math.floor(h / 24);
   138→  if (d < 30) return `${d} hari lalu`;
   139→  return fmtDate(iso);
   140→}

   142→function groupDeliverables(list: DeliverableWithProject[], projects: ProjectDTO[]): DeliverableGroup[] {
   143→  const projById = new Map(projects.map((p) => [p.id, p]));
   144→  const groups: DeliverableGroup[] = [];
   145→  for (const d of list) {
   146→    let g = groups.find((x) => x.projectId === d.projectId);
   147→    if (!g) {
   148→      const proj = projById.get(d.projectId);
   149→      g = {
   150→        projectId: d.projectId,
   151→        code: d.projectCode,
   152→        name: d.projectName,
   153→        status: proj?.status ?? null,
   154→        items: [],
   155→      };
   156→      groups.push(g);
   157→    }
   158→    g.items.push(d);
   159→  }
   160→  return groups;
   161→}

   514→                      aria-label={`Tambah file atau tautan untuk milestone ${ms.title}`}
   515→                      className="shrink-0"
   516→                      onClick={() => onOpenDialog(project.id, ms.title)}
   517→                    >
   518→                      <Plus className="size-3.5" aria-hidden />
   519→                    </Button>
   520→                  )}
   521→                </div>
   522→                {items.length > 0 ? (
   523→                  <div className="divide-y">
   524→                    {items.map((d) => (
   525→                      <DeliverableRow
   526→                        key={d.id}
   527→                        d={d}
   528→                        nowMs={nowMs}
   529→                        canDelete={canDeleteDeliverable(d)}
   530→                        busy={busyDelKey === d.id}
   531→                        onRemove={onRemoveDeliverable}
   532→                      />
   533→                    ))}
   534→                  </div>
   535→                ) : (
   536→                  <p className="px-4 py-2.5 text-xs text-muted-foreground">
   537→                    Belum ada file/tautan untuk milestone ini.
   538→                    {canSendDeliverable && " Klik tombol + di atas untuk menambahkan."}
   539→                  </p>
   540→                )}
   541→              </div>
   542→            );
   543→          })}
   544→
   545→          {/* Deliverable tanpa milestone */}
   546→          {loose.length > 0 && (
   547→            <div className="overflow-hidden rounded-xl border border-dashed">
   548→              <div className="flex items-center justify-between border-b bg-stone-50/70 px-3 py-2">
   549→                <span className="text-xs font-medium text-slate-600">Tanpa milestone</span>
   550→                <Badge variant="outline" className="px-1.5 py-0 text-[10px]">
   551→                  {loose.length} file
   552→                </Badge>
   553→              </div>
   554→              <div className="divide-y">
   555→                {loose.map((d) => (
   556→                  <DeliverableRow
   557→                    key={d.id}
   558→                    d={d}

   560→                    canDelete={canDeleteDeliverable(d)}
   561→                    busy={busyDelKey === d.id}
   562→                    onRemove={onRemoveDeliverable}
   563→                  />
   564→                ))}
   565→              </div>
   566→            </div>
   567→          )}
   568→        </CardContent>
   569→      </Card>
   570→    </div>
   571→  );
