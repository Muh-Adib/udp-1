    52→function formatRupiah(n: number): string {
    53→  return `Rp ${numberFmt.format(Number.isFinite(n) ? Math.round(n) : 0)}`;

    56→function formatDate(iso?: string | null): string {
    57→  if (!iso) return "—";
    58→  return new Date(iso).toLocaleDateString("id-ID", { day: "numeric", month: "short", year: "numeric" });
    59→}

    61→/** Lewat deadline = tanggal deadline sebelum hari ini & brief belum ditawarkan. */
    62→function isOverdue(deadline: string, status: BriefDTO["status"], nowTs: number | null): boolean {
    63→  if (!nowTs || status === "QUOTED") return false;
    64→  const startOfToday = new Date(nowTs);
    65→  startOfToday.setHours(0, 0, 0, 0);
    66→  return new Date(deadline).getTime() < startOfToday.getTime();
    67→}

   112→

   114→  const isUrl = (s: string) => /^https?:\/\//i.test(s) || /^www\./i.test(s);

   117→  const showEstimate = brief.status !== "QUOTED" && canEstimate;

   122→    <Card className="flex flex-col rounded-2xl transition-shadow hover:shadow-md">
   123→      <CardContent className="flex flex-1 flex-col gap-2.5 pt-5">
   124→        {/* kode + status + brand */}

   128→            {BRIEF_STATUS_LABEL[brief.status]}
   129→          </Badge>
   130→          <Badge variant="outline" className="ml-auto border-stone-200 bg-stone-100 text-stone-700">
   131→            {BRAND_LABEL[brief.brand] ?? brief.brand}
   132→          </Badge>
   133→        </div>
   134→
   135→        {/* judul */}
   136→        <p className="text-sm font-bold leading-snug text-slate-900">{brief.title}</p>
   137→
   138→        {/* info lead */}
   139→        {lead && (
   140→          <div className="flex flex-wrap items-center gap-x-1.5 gap-y-1 text-xs text-muted-foreground">
   141→            <span className="font-mono">{lead.code}</span>
   142→            <span aria-hidden="true">·</span>
   143→            <span className="font-medium text-slate-700">{lead.contactName}</span>
   144→            {lead.companyName && (
   145→              <>
   146→                <span aria-hidden="true">·</span>
   147→                <span className="truncate">{lead.companyName}</span>
   148→              </>
   149→            )}
   150→            {lead.stage && (
   151→              <Badge variant="outline" className={cn("px-1.5 py-0 text-[10px]", LEAD_STAGE_BADGE[lead.stage as LeadStage])}>
   152→                {LEAD_STAGE_LABEL[lead.stage as LeadStage]}
   153→              </Badge>
   154→            )}
   155→          </div>
   156→        )}
   157→
   158→        {/* tujuan */}
   159→        <p className="line-clamp-2 text-xs leading-relaxed text-slate-600">
   160→          <span className="font-semibold text-slate-800">Tujuan: </span>
   161→          {brief.objective}
   162→        </p>
   163→
   164→        {/* deadline */}
   165→        {brief.deadline && (
   166→          <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   167→            <span className="inline-flex items-center gap-1">
   168→              <CalendarDays className="size-3.5" /> Deadline {formatDate(brief.deadline)}
   169→            </span>
   170→            {overdue && <Badge className="border-rose-200 bg-rose-100 text-rose-700">Lewat deadline</Badge>}
   171→          </div>
   172→        )}
   173→
   174→        {/* deliverables (maks 3 baris) */}
   175→        {shownDeliverables.length > 0 && (
   176→          <ul className="space-y-1">
   177→            {shownDeliverables.map((d, i) => (
   178→              <li key={i} className="flex items-start gap-1.5 text-xs text-slate-600">
   179→                <CheckCircle2 className="mt-0.5 size-3.5 shrink-0 text-emerald-600" aria-hidden="true" />
   180→                <span className="min-w-0">{d}</span>
   181→              </li>
   182→            ))}
   183→            {restDeliverables > 0 && (
   184→              <li className="pl-5 text-xs text-muted-foreground">+{restDeliverables} lainnya</li>
   185→            )}
   186→          </ul>
   187→        )}
   188→
   189→        {/* referensi */}
   190→        {refLines.length > 0 && (
   191→          <div className="flex flex-wrap items-center gap-x-3 gap-y-1">
   192→            {refLines.map((ref, i) =>
   193→              isUrl(ref) ? (
   194→                <a
   195→                  key={i}
   196→                  href={ref.startsWith("www.") ? `https://${ref}` : ref}
   197→                  target="_blank"
   198→                  rel="noopener noreferrer"
   199→                  className="inline-flex items-center gap-1 text-xs font-medium text-teal-700 hover:underline"
   200→                >
   201→                  <Link2 className="size-3" aria-hidden="true" /> Referensi{refLines.length > 1 ? ` ${i + 1}` : ""}
   202→                </a>
   203→              ) : (
   204→                <span key={i} className="inline-flex items-center gap-1 truncate text-xs text-muted-foreground">
   205→                  <Link2 className="size-3 shrink-0" aria-hidden="true" /> {ref}
   206→                </span>
   207→              )
   208→            )}
   209→          </div>
   210→        )}
   211→
   212→        {/* estimasi produksi (accordion) */}
   213→        {brief.estimates.map((est) => {
   214→          const open = openEstimateId === est.id;
   215→          return (
   216→            <div key={est.id} className="rounded-xl border border-teal-200 bg-teal-50/70 p-3">
   217→              <button
   218→                type="button"
   219→                onClick={() => onToggleEstimate(est.id)}
   220→                aria-expanded={open}
   221→                className="flex w-full items-center gap-2 text-left"
   222→              >
   223→                <Clock3 className="size-4 shrink-0 text-teal-600" aria-hidden="true" />
   224→                <span className="text-sm font-semibold text-teal-900">
   225→                  Estimasi: {est.totalHours} jam · {formatRupiah(est.totalCost)}
   226→                </span>
   227→                <ChevronDown
   228→                  className={cn("ml-auto size-4 shrink-0 text-teal-600 transition-transform", open && "rotate-180")}
   229→                  aria-hidden="true"
   230→                />
   231→              </button>
   232→              {est.createdByName && <p className="mt-0.5 pl-6 text-[11px] text-teal-700">Dihitung oleh {est.createdByName}</p>}
   233→              {open && (
   234→                <div className="mt-2.5 space-y-1.5 border-t border-teal-200/80 pt-2.5">
   235→                  {est.items.map((item, i) => (
   236→                    <div key={i} className="flex items-baseline justify-between gap-2 text-xs">
   237→                      <span className="min-w-0 flex-1 truncate text-slate-700" title={item.task}>
   238→                        {item.task}
   239→                      </span>
   240→                      <span className="shrink-0 whitespace-nowrap text-muted-foreground">
   241→                        {item.qty} {item.unit} · {item.hours} jam
   242→                      </span>
   243→                      <span className="w-24 shrink-0 text-right font-medium text-teal-900">
   244→                        {formatRupiah(item.qty * item.cost)}
   245→                      </span>
   246→                    </div>
   247→                  ))}
   248→                  {est.notes && <p className="pt-1 text-xs italic text-teal-800/80">{est.notes}</p>}
   249→                </div>
   250→              )}
   251→            </div>
   252→          );
   253→        })}
   254→
   255→        {/* aksi per role */}
   256→        {hasFooter && (
   257→          <div className="mt-auto flex flex-wrap items-center gap-2 border-t border-border/60 pt-3">
   258→            {brief.projectCode && (
   259→              <Badge className="border-emerald-200 bg-emerald-100 text-emerald-800">
   260→                <FolderKanban className="size-3" aria-hidden="true" /> Proyek: {brief.projectCode}
   261→              </Badge>
   262→            )}
   263→            {showSubmit && (
   264→              <Button size="sm" variant="outline" onClick={() => onAskSubmit(brief)}>
   265→                <Send className="size-3.5" /> Kirim untuk Estimasi
   266→              </Button>
   267→            )}
   268→            {showEstimate && (
   269→              <Button size="sm" onClick={() => onOpenEstimate(brief)}>
   270→                <Calculator className="size-3.5" /> Buat Estimasi
   271→              </Button>
   272→            )}
   273→            {showHint && (
   274→              <span className="inline-flex items-center gap-1.5 rounded-lg border border-emerald-200 bg-emerald-50 px-2.5 py-1.5 text-xs font-medium text-emerald-800">
   275→                <CheckCircle2 className="size-3.5" aria-hidden="true" /> Siap dibuatkan penawaran di menu Keuangan
   276→              </span>
   277→            )}
   278→          </div>
   279→        )}
   280→      </CardContent>
   281→    </Card>
   282→  );
   283→}

   314→
   315→  // dialog kirim untuk estimasi

   318→

   321→  const [estRows, setEstRows] = useState<EstimateRow[]>([emptyEstimateRow()]);
   322→  const [estNotes, setEstNotes] = useState("");
   323→
   324→  const role = user.role;

   326→  const canSubmit = canCreateBrief;

   328→

   353→      const { briefs: data } = await api.briefs();

   440→      toast.error((e as Error).message);

   442→      setBusy(null);

   446→  /* ---------- dialog buat estimasi ---------- */

   448→  function openEstimate(brief: BriefDTO) {
   449→    setEstimateTarget(brief);
   450→    setEstRows([emptyEstimateRow()]);

   457→  async function submitEstimate() {
   458→    const target = estimateTarget;
   459→    if (!target) return;
   460→    const items: EstimateItemDTO[] = estRows
   461→      .map((r) => ({
   462→        task: r.task.trim(),
   463→        qty: Math.max(0, Math.round(Number(r.qty) || 0)),
   464→        unit: r.unit,
   465→        hours: Math.max(0, Math.round(Number(r.hours) || 0)),
   466→        cost: Math.max(0, Math.round(Number(r.cost) || 0)),
   467→      }))
   468→      .filter((it) => it.task && it.qty > 0);
   469→    if (items.length === 0) {
   470→      toast.error("Minimal satu baris pekerjaan valid (nama pekerjaan + qty > 0)");
   471→      return;
   472→    }
   473→    setBusy("estimate");
   474→    try {
   475→      const { estimate } = await api.createEstimate({
   476→        briefId: target.id,
   477→        items,
   478→        notes: estNotes.trim() || undefined,
   479→      });
   480→      toast.success(
   481→        `Estimasi ${target.code} tersimpan — ${estimate.totalHours} jam · ${formatRupiah(estimate.totalCost)}`
   482→      );
   483→      setEstimateTarget(null);
   484→      await refreshData();
   485→    } catch (e) {
   486→      toast.error((e as Error).message);
   487→    } finally {
   488→      setBusy(null);
   489→    }
   490→  }
   491→
   492→  /* ---------- render: error ---------- */
   493→
   494→  if (error && !briefs) {
   495→    return (
   496→      <Card className="mx-auto max-w-md border-rose-200">
   497→        <CardContent className="flex flex-col items-center gap-3 py-10 text-center">
   498→          <span className="flex size-12 items-center justify-center rounded-2xl bg-rose-100 text-rose-600">
   499→            <TriangleAlert className="size-6" />
   500→          </span>
   501→          <div>
   502→            <p className="font-semibold">Gagal memuat data brief</p>
   503→            <p className="text-sm text-muted-foreground">{error}</p>
   504→          </div>
   505→          <Button onClick={() => void load()}>
   506→            <RefreshCw className="size-4" /> Coba Lagi
   507→          </Button>
   508→        </CardContent>
   509→      </Card>
   510→    );
   511→  }
   512→
   513→  /* ---------- render: skeleton ---------- */
   514→
   515→  if (loading || !briefs) {
   516→    return (
   517→      <div className="space-y-5">
   518→        <Card>
   519→          <CardContent className="animate-pulse space-y-3 pt-5">
   520→            <div className="h-6 w-56 rounded bg-muted" />
   521→            <div className="h-3 w-80 max-w-full rounded bg-muted" />
   522→          </CardContent>
