    81→/** Label pemisah tanggal: Hari Ini / Kemarin / tanggal singkat. */
    82→const dayLabel = (iso: string) => {
    83→  const d = new Date(iso)
    84→  const now = new Date()
    85→  const yesterday = new Date(now)
    86→  yesterday.setDate(now.getDate() - 1)
    87→  if (sameDay(d, now)) return 'Hari Ini'
    88→  if (sameDay(d, yesterday)) return 'Kemarin'
    89→  return formatDate(iso)
    90→}
