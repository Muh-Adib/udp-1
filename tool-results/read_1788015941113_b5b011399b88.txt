     2→'use client'
     3→
     4→import React, { useCallback, useEffect, useMemo, useRef, useState } from 'react'
     5→import { cn } from '@/lib/utils'
     6→import { useToast } from '@/hooks/use-toast'
     7→import {
     8→  Sheet, SheetContent, SheetClose, SheetTitle, SheetDescription,
     9→} from '@/components/ui/sheet'
    10→import { Badge } from '@/components/ui/badge'
    11→import { Button } from '@/components/ui/button'
    12→import { Input } from '@/components/ui/input'
    13→import { Textarea } from '@/components/ui/textarea'
    14→import { Checkbox } from '@/components/ui/checkbox'
    15→import { Progress } from '@/components/ui/progress'
    16→import { Alert, AlertDescription, AlertTitle } from '@/components/ui/alert'
    17→import { Skeleton } from '@/components/ui/skeleton'
    18→import { Separator } from '@/components/ui/separator'
    19→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    20→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    21→import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from '@/components/ui/dialog'
    22→import { DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuSeparator, DropdownMenuTrigger } from '@/components/ui/dropdown-menu'
    23→import {
    24→  BrandChip, StageBadge, PriorityBadge, TempBadge, ChannelIcon, UserAvatar, EmptyState,
    25→} from './shared'
    26→import { crmApi } from './api-client'
    27→import { useCrmStore } from './crm-store'
    28→import type { OpportunityDetailDTO, Stage, Temperature, Priority, ContactDTO } from '@/lib/crm-types'
    29→import {
    30→  STAGES, stageMeta, LOST_REASONS, lostReasonLabel, REACTIVATION_OPTIONS, reactivationLabel,
    31→  channelMeta, LEAD_SOURCES, leadSourceLabel, PRIORITIES, priorityMeta, TEMPERATURES, temperatureMeta,
    32→  TASK_TYPES, projectStatusMeta, formatMoney, formatDate, timeAgo,
    33→} from '@/lib/crm-constants'
    34→import {
    35→  Building2, User, Mail, MessageCircle, MoreVertical, AlertTriangle, CheckCircle2, Circle,
    36→  Clock, Send, Trash2, Settings2, CalendarClock, Paperclip, Link2, CornerDownRight, X,
    37→} from 'lucide-react'
    38→
    39→const SCROLLBAR = '[&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-slate-300'
    40→
    41→type TabKey = 'ringkasan' | 'percakapan' | 'tugas' | 'catatan'
    42→
    43→interface LostForm { reason: string; notes: string; competitor: string; reactivation: string; followUpDate: string }
    44→
    45→const EMPTY_LOST: LostForm = { reason: '', notes: '', competitor: '', reactivation: '', followUpDate: '' }
    46→
    47→/* ---------- Small definition item ---------- */
    48→function DefItem({ label, value }: { label: string; value?: React.ReactNode | null }) {
    49→  const empty = value === null || value === undefined || value === ''
    50→  return (
    51→    <div className="min-w-0">
    52→      <p className="text-[11px] font-semibold uppercase tracking-wide text-slate-400">{label}</p>
    53→      <div className={cn('mt-0.5 whitespace-pre-wrap break-words text-sm', empty ? 'text-slate-300' : 'text-slate-700')}>
    54→        {empty ? '—' : value}
    55→      </div>
    56→    </div>
    57→  )
    58→}
    59→
    60→export function OpportunityDetailDrawer({ opportunityId, open, onClose, onChanged }: {
    61→  opportunityId: string | null
    62→  open: boolean
    63→  onClose: () => void
    64→  onChanged?: () => void
    65→}) {
    66→  const { toast } = useToast()
    67→  const { user, openOpportunity } = useCrmStore()
    68→
    69→  const [detail, setDetail] = useState<OpportunityDetailDTO | null>(null)
    70→  const [contact, setContact] = useState<ContactDTO | null>(null)
    71→  const [loading, setLoading] = useState(false)
    72→  const [error, setError] = useState<string | null>(null)
    73→  const [tab, setTab] = useState<TabKey>('ringkasan')
    74→
    75→  /* composer */
    76→  const composerRef = useRef<HTMLDivElement | null>(null)
    77→  const [composerChannel, setComposerChannel] = useState('WHATSAPP')
    78→  const [composerBody, setComposerBody] = useState('')
    79→  const [sending, setSending] = useState(false)
    80→
    81→  /* task form */
    82→  const [taskTitle, setTaskTitle] = useState('')
    83→  const [taskAssignee, setTaskAssignee] = useState('')
    84→  const [taskDue, setTaskDue] = useState('')
    85→  const [taskType, setTaskType] = useState('FOLLOW_UP')
    86→  const [addingTask, setAddingTask] = useState(false)
    87→  const [togglingTask, setTogglingTask] = useState<string | null>(null)
    88→
    89→  /* note form */
    90→  const [noteBody, setNoteBody] = useState('')
    91→  const [noteVisibility, setNoteVisibility] = useState('INTERNAL')
    92→  const [addingNote, setAddingNote] = useState(false)
    93→
    94→  /* dialogs */
    95→  const [movingStage, setMovingStage] = useState(false)
    96→  const [lostOpen, setLostOpen] = useState(false)
    97→  const [lostForm, setLostForm] = useState<LostForm>(EMPTY_LOST)
    98→  const [wonOpen, setWonOpen] = useState(false)
    99→  const [wonOffer, setWonOffer] = useState('')
   100→  const [priorityOpen, setPriorityOpen] = useState(false)
   101→  const [priPriority, setPriPriority] = useState<Priority>('MEDIUM')
   102→  const [priTemperature, setPriTemperature] = useState<Temperature>('WARM')
   103→  const [nextActionOpen, setNextActionOpen] = useState(false)
   104→  const [naText, setNaText] = useState('')
   105→  const [naDate, setNaDate] = useState('')
   106→  const [deleteOpen, setDeleteOpen] = useState(false)
   107→  const [deleting, setDeleting] = useState(false)
   108→
   109→  const users = useCrmStore((s) => s.users)
   110→  const activeUsers = useMemo(() => users.filter((u) => u.isActive), [users])
   111→
   112→  /* ---------- data ---------- */
   113→  const load = useCallback(async () => {
   114→    if (!opportunityId) return
   115→    setLoading(true)
   116→    setError(null)
   117→    try {
   118→      const d = await crmApi.opportunity(opportunityId)
   119→      setDetail(d)
   120→      try {
   121→        const cs = await crmApi.contacts(`companyId=${d.companyId}`)
   122→        setContact(cs.find((c) => c.id === d.contactId) ?? null)
   123→      } catch { setContact(null) }
   124→    } catch (e) {
   125→      setError(e instanceof Error ? e.message : 'Gagal memuat detail opportunity')
   126→    } finally {
   127→      setLoading(false)
   128→    }
   129→  }, [opportunityId])
   130→
   131→  useEffect(() => {
   132→    if (open && opportunityId) {
   133→      void load()
   134→    }
   135→    if (!open) {
   136→      setDetail(null)
   137→      setContact(null)
   138→      setError(null)
   139→      setTab('ringkasan')
   140→      setComposerBody('')
   141→      setComposerChannel('WHATSAPP')
   142→      setTaskTitle(''); setTaskDue(''); setTaskType('FOLLOW_UP'); setTaskAssignee('')
   143→      setNoteBody(''); setNoteVisibility('INTERNAL')
   144→      setLostForm(EMPTY_LOST); setWonOffer('')
   145→    }
   146→  }, [open, opportunityId, load])
   147→
   148→  /* ---------- stage mutations ---------- */
   149→  const doChangeStage = useCallback(async (stage: Stage, extra?: Record<string, unknown>, successTitle?: string) => {
   150→    if (!detail) return
   151→    setMovingStage(true)
   152→    try {
   153→      await crmApi.changeStage(detail.id, { stage, ...extra })
   154→      toast({ title: successTitle ?? 'Stage diperbarui', description: `${detail.code} dipindah ke ${stageMeta(stage).label}` })
   155→      setLostOpen(false); setWonOpen(false)
   156→      await load()
   157→      onChanged?.()
   158→    } catch (e) {
   159→      toast({ title: 'Gagal memindah stage', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   160→    } finally {
   161→      setMovingStage(false)
   162→    }
   163→  }, [detail, load, onChanged, toast])
   164→
   165→  const handleStageIntent = (target: string) => {
   166→    if (!detail || target === detail.stage) return
   167→    if (target === 'LOST') { setLostForm(EMPTY_LOST); setLostOpen(true); return }
   168→    if (target === 'WON') { setWonOffer(''); setWonOpen(true); return }
   169→    void doChangeStage(target as Stage)
   170→  }
   171→
   172→  const submitLost = () => {
   173→    if (!lostForm.reason) return
   174→    void doChangeStage('LOST', {
   175→      lostReason: lostForm.reason,
   176→      ...(lostForm.notes.trim() ? { lostNotes: lostForm.notes.trim() } : {}),
   177→      ...(lostForm.competitor.trim() ? { competitorName: lostForm.competitor.trim() } : {}),
   178→      ...(lostForm.reactivation ? { reactivation: lostForm.reactivation } : {}),
   179→      ...(lostForm.followUpDate ? { followUpDate: lostForm.followUpDate } : {}),
   180→    }, 'Opportunity ditandai Lost')
   181→  }
   182→
   183→  const confirmWon = () => {
   184→    const offer = Number(wonOffer)
   185→    void doChangeStage('WON', offer > 0 ? { lastOfferValue: offer } : {}, '🎉 Deal Won! Project produksi dibuat otomatis')
   186→  }
   187→
   188→  /* ---------- priority / next action / delete ---------- */
   189→  const openPriorityDialog = () => {
   190→    if (!detail) return
   191→    setPriPriority(detail.priority)
   192→    setPriTemperature(detail.temperature)
   193→    setPriorityOpen(true)
   194→  }
   195→
   196→  const savePriority = async () => {
   197→    if (!detail) return
   198→    if (priPriority === detail.priority && priTemperature === detail.temperature) { setPriorityOpen(false); return }
   199→    try {
   200→      await crmApi.updateOpportunity(detail.id, { priority: priPriority, temperature: priTemperature })
   201→      toast({ title: 'Prioritas diperbarui', description: `${detail.code} → ${priorityMeta(priPriority).label} · ${temperatureMeta(priTemperature).label}` })
   202→      setPriorityOpen(false)
   203→      await load()
   204→      onChanged?.()
   205→    } catch (e) {
   206→      toast({ title: 'Gagal memperbarui prioritas', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   207→    }
   208→  }
   209→
   210→  const openNextActionDialog = () => {
   211→    if (!detail) return
   212→    setNaText(detail.nextAction ?? '')
   213→    setNaDate(detail.nextActionDate ? detail.nextActionDate.slice(0, 10) : '')
   214→    setNextActionOpen(true)
   215→  }
   216→
   217→  const saveNextAction = async () => {
   218→    if (!detail) return
   219→    const text = naText.trim()
   220→    const same = text === (detail.nextAction ?? '') && naDate === (detail.nextActionDate ? detail.nextActionDate.slice(0, 10) : '')
   221→    if (same) { setNextActionOpen(false); return }
   222→    try {
   223→      await crmApi.updateOpportunity(detail.id, { nextAction: text, nextActionDate: naDate })
   224→      toast({ title: 'Next action diperbarui' })
   225→      setNextActionOpen(false)
   226→      await load()
   227→      onChanged?.()
   228→    } catch (e) {
   229→      toast({ title: 'Gagal memperbarui next action', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   230→    }
   231→  }
   232→
   233→  const confirmDelete = async () => {
   234→    if (!detail) return
   235→    setDeleting(true)
   236→    try {
   237→      await crmApi.deleteOpportunity(detail.id)
   238→      toast({ title: 'Opportunity dihapus', description: `${detail.code} dihapus dari pipeline` })
   239→      setDeleteOpen(false)
   240→      onClose()
   241→      onChanged?.()
   242→    } catch (e) {
   243→      toast({ title: 'Gagal menghapus', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   244→    } finally {
   245→      setDeleting(false)
   246→    }
   247→  }
   248→
   249→  /* ---------- interactions ---------- */
   250→  const scrollToComposer = () => {
   251→    setTab('percakapan')
   252→    setTimeout(() => composerRef.current?.scrollIntoView({ behavior: 'smooth', block: 'center' }), 80)
   253→  }
   254→
   255→  const sendInteraction = async () => {
   256→    if (!detail || !composerBody.trim()) return
   257→    setSending(true)
   258→    try {
   259→      await crmApi.addInteraction(detail.id, { channel: composerChannel, direction: 'OUT', body: composerBody.trim() })
   260→      toast({ title: 'Terkirim — stage otomatis naik ke Connected bila sebelumnya New' })
   261→      setComposerBody('')
   262→      await load()
   263→      onChanged?.()
   264→    } catch (e) {
   265→      toast({ title: 'Gagal mengirim', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   266→    } finally {
   267→      setSending(false)
   268→    }
   269→  }
   270→
   271→  /* ---------- tasks ---------- */
   272→  const toggleTask = async (taskId: string, done: boolean) => {
   273→    setTogglingTask(taskId)
   274→    try {
   275→      await crmApi.updateTask(taskId, { status: done ? 'DONE' : 'OPEN' })
   276→      await load()
   277→      onChanged?.()
   278→    } catch (e) {
   279→      toast({ title: 'Gagal memperbarui task', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   280→    } finally {
   281→      setTogglingTask(null)
   282→    }
   283→  }
   284→
   285→  const submitTask = async () => {
   286→    if (!detail) return
   287→    if (!taskTitle.trim() || !taskAssignee) {
   288→      toast({ title: 'Lengkapi judul dan assignee task', variant: 'destructive' })
   289→      return
   290→    }
   291→    setAddingTask(true)
   292→    try {
   293→      await crmApi.createTask({
   294→        opportunityId: detail.id,
   295→        title: taskTitle.trim(),
   296→        assigneeId: taskAssignee,
   297→        type: taskType,
   298→        ...(taskDue ? { dueDate: taskDue } : {}),
   299→      })
   300→      toast({ title: 'Task ditambahkan' })
   301→      setTaskTitle(''); setTaskDue(''); setTaskType('FOLLOW_UP')
   302→      await load()
   303→      onChanged?.()
   304→    } catch (e) {
   305→      toast({ title: 'Gagal menambah task', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   306→    } finally {
   307→      setAddingTask(false)
   308→    }
   309→  }
   310→
   311→  /* ---------- notes ---------- */
   312→  const submitNote = async () => {
   313→    if (!detail || !noteBody.trim()) return
   314→    setAddingNote(true)
   315→    try {
   316→      await crmApi.addNote(detail.id, { body: noteBody.trim(), visibility: noteVisibility })
   317→      toast({ title: 'Catatan disimpan' })
   318→      setNoteBody('')
   319→      await load()
   320→      onChanged?.()
   321→    } catch (e) {
   322→      toast({ title: 'Gagal menyimpan catatan', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   323→    } finally {
   324→      setAddingNote(false)
   325→    }
   326→  }
   327→
   328→  /* ---------- derived ---------- */
   329→  const overdueTasks = useMemo(() => {
   330→    if (!detail) return []
   331→    const now = Date.now()
   332→    return detail.tasks.filter((t) => t.status !== 'DONE' && t.dueDate && new Date(t.dueDate).getTime() < now)
   333→  }, [detail])
   334→
   335→  /* ---------- render ---------- */
   336→  return (
   337→    <Sheet open={open} onOpenChange={(o) => { if (!o) onClose() }}>
   338→      <SheetContent side="right" className={cn('flex w-full flex-col gap-0 overflow-y-auto p-0 sm:max-w-2xl [&>button]:hidden', SCROLLBAR)}>
   339→        <SheetTitle className="sr-only">Detail Opportunity</SheetTitle>
   340→        <SheetDescription className="sr-only">Ringkasan lengkap opportunity, percakapan, tugas, dan catatan</SheetDescription>
   341→
   342→        {/* loading / error / empty states */}
   343→        {loading && !detail && (
   344→          <div className="space-y-4 p-5">
   345→            <Skeleton className="h-4 w-24" />
   346→            <Skeleton className="h-7 w-3/4" />
   347→            <div className="flex gap-2"><Skeleton className="h-6 w-20" /><Skeleton className="h-6 w-24" /><Skeleton className="h-6 w-16" /></div>
   348→            <Skeleton className="h-8 w-56" />
   349→            <Skeleton className="h-24 w-full rounded-xl" />
   350→            <Skeleton className="h-64 w-full rounded-xl" />
   351→          </div>
   352→        )}
   353→
   354→        {!loading && error && (
   355→          <div className="p-5">
   356→            <Alert variant="destructive">
   357→              <AlertTriangle className="h-4 w-4" />
   358→              <AlertTitle>Gagal memuat opportunity</AlertTitle>
   359→              <AlertDescription>{error}</AlertDescription>
   360→            </Alert>
   361→            <Button variant="outline" size="sm" className="mt-3" onClick={() => void load()}>Coba lagi</Button>
   362→          </div>
   363→        )}
   364→
   365→        {detail && (
   366→          <>
   367→            {/* ---------- sticky header ---------- */}
   368→            <div className="sticky top-0 z-20 border-b border-slate-100 bg-white/95 px-5 pb-4 pt-4 backdrop-blur">
   369→              <div className="flex items-start justify-between gap-2 pr-8">
   370→                <div className="min-w-0">
   371→                  <p className="font-mono text-xs text-slate-400">{detail.code}</p>
   372→                  <h2 className="mt-0.5 text-lg font-bold leading-snug text-slate-900">{detail.title}</h2>
   373→                </div>
   374→                <SheetClose asChild>
   375→                  <Button variant="ghost" size="icon" className="h-8 w-8 shrink-0 text-slate-400 hover:text-slate-700" title="Tutup">
   376→                    <X className="h-4 w-4" />
   377→                  </Button>
   378→                </SheetClose>
   379→              </div>
   380→
   381→              <div className="mt-2 flex flex-wrap items-center gap-2">
   382→                <BrandChip name={detail.brandName} color={detail.brandColor} />
   383→                <StageBadge stage={detail.stage} />
   384→                <TempBadge temperature={detail.temperature} />
   385→                <PriorityBadge priority={detail.priority} />
   386→                <span className="rounded-md bg-slate-100 px-2 py-0.5 text-[11px] font-bold text-slate-600">{detail.probability}%</span>
   387→              </div>
   388→
   389→              <div className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-xs text-slate-500">
   390→                <span className="inline-flex items-center gap-1.5"><Building2 className="h-3.5 w-3.5 text-slate-400" />{detail.companyName}</span>
   391→                <span className="inline-flex items-center gap-1.5"><User className="h-3.5 w-3.5 text-slate-400" />{detail.contactName}</span>
   392→                {contact?.whatsapp && (
   393→                  <span className="inline-flex items-center gap-1 text-[11px] text-slate-400"><MessageCircle className="h-3 w-3" />{contact.whatsapp}</span>
   394→                )}
   395→                {contact?.email && (
   396→                  <span className="inline-flex items-center gap-1 text-[11px] text-slate-400"><Mail className="h-3 w-3" />{contact.email}</span>
   397→                )}
   398→              </div>
   399→
   400→              <div className="mt-3 flex flex-wrap items-end justify-between gap-3">
   401→                <div>
   402→                  <p className="text-xl font-extrabold tracking-tight text-slate-900">{formatMoney(detail.estimatedValue, detail.currency)}</p>
   403→                  <p className="inline-flex items-center gap-1 text-[11px] text-slate-400"><CalendarClock className="h-3 w-3" />Est. close: {formatDate(detail.expectedCloseDate)}</p>
   404→                </div>
   405→                <div className="flex flex-wrap items-center gap-2">
   406→                  <Button variant="outline" size="sm" className="gap-1.5" onClick={scrollToComposer}>✉️ Log Percakapan</Button>
   407→                  <Select value={detail.stage} onValueChange={handleStageIntent} disabled={movingStage}>
   408→                    <SelectTrigger size="sm" className="w-[168px]" title="Pindah stage">
   409→                      <SelectValue />
   410→                    </SelectTrigger>
   411→                    <SelectContent>
   412→                      {STAGES.map((s) => (
   413→                        <SelectItem key={s.key} value={s.key} disabled={s.key === detail.stage}>
   414→                          <span className="inline-flex items-center gap-2"><span className={cn('h-1.5 w-1.5 rounded-full', s.dot)} />{s.label}</span>
   415→                        </SelectItem>
   416→                      ))}
   417→                    </SelectContent>
   418→                  </Select>
   419→                  <DropdownMenu>
   420→                    <DropdownMenuTrigger asChild>
   421→                      <Button variant="ghost" size="icon" className="h-8 w-8 text-slate-500" title="Aksi lainnya">
   422→                        <MoreVertical className="h-4 w-4" />
   423→                      </Button>
   424→                    </DropdownMenuTrigger>
   425→                    <DropdownMenuContent align="end">
   426→                      <DropdownMenuItem onClick={openPriorityDialog}><Settings2 className="mr-2 h-4 w-4" />Edit Prioritas</DropdownMenuItem>
   427→                      <DropdownMenuItem onClick={openNextActionDialog}><CalendarClock className="mr-2 h-4 w-4" />Edit Next Action</DropdownMenuItem>
   428→                      <DropdownMenuSeparator />
   429→                      <DropdownMenuItem className="text-rose-600 focus:text-rose-700" onClick={() => setDeleteOpen(true)}>
   430→                        <Trash2 className="mr-2 h-4 w-4" />Hapus
   431→                      </DropdownMenuItem>
   432→                    </DropdownMenuContent>
   433→                  </DropdownMenu>
   434→                </div>
   435→              </div>
   436→            </div>
   437→
   438→            {/* ---------- tabs ---------- */}
   439→            <Tabs value={tab} onValueChange={(v) => setTab(v as TabKey)} className="flex min-h-0 flex-1 flex-col">
   440→              <div className="sticky top-0 z-10 bg-white px-5 pt-3">
   441→                <TabsList className="w-full justify-start overflow-x-auto">
   442→                  <TabsTrigger value="ringkasan">Ringkasan</TabsTrigger>
   443→                  <TabsTrigger value="percakapan" className="gap-1.5">Percakapan <span className="rounded-full bg-slate-100 px-1.5 text-[10px] font-semibold text-slate-500">{detail.interactions.length}</span></TabsTrigger>
   444→                  <TabsTrigger value="tugas" className="gap-1.5">Tugas <span className="rounded-full bg-slate-100 px-1.5 text-[10px] font-semibold text-slate-500">{detail.tasks.length}</span></TabsTrigger>
   445→                  <TabsTrigger value="catatan" className="gap-1.5">Catatan <span className="rounded-full bg-slate-100 px-1.5 text-[10px] font-semibold text-slate-500">{detail.notes.length}</span></TabsTrigger>
   446→                </TabsList>
   447→              </div>
   448→
   449→              {/* ===== Ringkasan ===== */}
   450→              <TabsContent value="ringkasan" className="space-y-5 px-5 pb-8 pt-4">
   451→                {overdueTasks.length > 0 && (
