   118→   118→function fmtDate(iso?: string | null): string {

   122→   122→function formatBytes(bytes: number): string {
   123→   123→  if (bytes >= 1024 * 1024) return `${(bytes / (1024 * 1024)).toFixed(1)} MB`;
   124→   124→  if (bytes >= 1024) return `${Math.round(bytes / 1024)} KB`;
   125→   125→  return `${bytes} B`;
   126→   126→}

   128→   128→/** Tanggal relatif id-ID (butuh `now` dari state agar purity-safe saat render). */
   129→   129→function timeAgoId(iso: string, now: number): string {
   130→   130→  if (!now) return fmtDate(iso);
   131→   131→  const diff = now - new Date(iso).getTime();
   132→   132→  if (diff < 60_000) return "baru saja";
   133→   133→  const m = Math.floor(diff / 60_000);
   134→   134→  if (m < 60) return `${m} menit lalu`;
   135→   135→  const h = Math.floor(m / 60);
   136→   136→  if (h < 24) return `${h} jam lalu`;
   137→   137→  const d = Math.floor(h / 24);
   138→   138→  if (d < 30) return `${d} hari lalu`;
   139→   139→  return fmtDate(iso);
   140→   140→}

   142→   142→function groupDeliverables(list: DeliverableWithProject[], projects: ProjectDTO[]): DeliverableGroup[] {
   143→   143→  const projById = new Map(projects.map((p) => [p.id, p]));
   144→   144→  const groups: DeliverableGroup[] = [];
   145→   145→  for (const d of list) {
   146→   146→    let g = groups.find((x) => x.projectId === d.projectId);
   147→   147→    if (!g) {
   148→   148→      const proj = projById.get(d.projectId);
   149→   149→      g = {
   150→   150→        projectId: d.projectId,
   151→   151→        code: d.projectCode,
   152→   152→        name: d.projectName,
   153→   153→        status: proj?.status ?? null,
   154→   154→        items: [],
   155→   155→      };
   156→   156→      groups.push(g);
   157→   157→    }
   158→   158→    g.items.push(d);
   159→   159→  }
   160→   160→  return groups;
   161→   161→}
