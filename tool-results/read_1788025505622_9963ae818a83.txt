   178→/* ---------- Kanban card ---------- */
   179→function PipelineCard({ o, busy, onOpen, onMove }: {
   180→  o: OpportunityDTO
   181→  busy: boolean
   182→  onOpen: () => void
   183→  onMove: (stage: Stage) => void
   184→}) {
   185→  const isClosed = o.stage === 'WON' || o.stage === 'LOST'
   186→  const overdue = !isClosed && !!o.nextActionDate && new Date(o.nextActionDate).getTime() < Date.now()
   187→  const suppressClick = React.useRef(false)
   188→  return (
   189→    <div
   190→      draggable
   191→      onDragStart={(e) => { e.dataTransfer.setData('text/plain', o.id); e.dataTransfer.effectAllowed = 'move' }}
   192→      onClick={() => {
   193→        if (suppressClick.current) { suppressClick.current = false; return }
   194→        onOpen()
   195→      }}
   196→      className={cn(
   197→        'group cursor-pointer rounded-xl border border-slate-200 border-l-4 bg-white p-3 shadow-sm transition hover:shadow-md',
   198→        busy && 'pointer-events-none opacity-50',
   199→      )}
   200→      style={{ borderLeftColor: o.brandColor }}
   201→    >
   202→      <div className="flex items-start gap-1">
   203→        <div className="min-w-0 flex-1">
   204→          <p className="line-clamp-1 text-[13px] font-semibold text-slate-800">{o.title}</p>
   205→          <p className="text-[10px] text-slate-400">{o.code}</p>
   206→        </div>
   207→        <DropdownMenu>
   208→          <DropdownMenuTrigger asChild>
   209→            <button
   210→              onClick={(e) => e.stopPropagation()}
   211→              className="rounded p-0.5 text-slate-300 transition hover:bg-slate-100 hover:text-slate-600 sm:opacity-0 sm:group-hover:opacity-100"
   212→              title="Pindah ke…"
   213→            >
   214→              <ChevronDown className="h-3.5 w-3.5" />
   215→            </button>
   216→          </DropdownMenuTrigger>
   217→          <DropdownMenuContent align="end" className="max-h-72 overflow-y-auto">
   218→            <DropdownMenuLabel className="text-[11px] text-slate-400">Pindah ke…</DropdownMenuLabel>
   219→            {STAGES.filter((s) => s.key !== o.stage).map((s) => (
   220→              <DropdownMenuItem key={s.key} onSelect={() => { suppressClick.current = true; onMove(s.key) }}>
   221→                <span className={cn('mr-2 h-1.5 w-1.5 rounded-full', s.dot)} />{s.label}
   222→              </DropdownMenuItem>
   223→            ))}
   224→          </DropdownMenuContent>
   225→        </DropdownMenu>
   226→      </div>
   227→
   228→      <p className="truncate text-xs text-slate-500">{o.companyName} · {o.contactName}</p>
   229→
   230→      <div className="mt-2 flex items-center gap-1.5">
   231→        <BrandChip name={o.brandName} color={o.brandColor} size="xs" />
   232→        <TempIcon temperature={o.temperature} />
   233→        {o.serviceName && <span className="truncate text-[10px] text-slate-400">· {o.serviceName}</span>}
   234→      </div>
   235→
   236→      <div className="mt-2 flex items-center justify-between gap-2">
   237→        <span className="text-sm font-bold text-slate-900">{formatMoney(o.estimatedValue, o.currency, true)}</span>
   238→        <div className="flex items-center gap-1">
   239→          {!isClosed && <ScoreBadge result={computeLeadScore(o)} compact />}
   240→          <span className="rounded bg-slate-100 px-1.5 py-0.5 text-[10px] font-bold text-slate-500">{o.probability}%</span>
   241→        </div>
   242→      </div>
   243→
   244→      {o.nextAction && (
   245→        <p className="mt-2 truncate rounded bg-amber-50 px-1.5 py-0.5 text-[11px] text-amber-700" title={o.nextAction}>
   246→          {o.nextAction}
   247→          {overdue && (
   248→            <span className="ml-1 inline-flex items-center gap-0.5 whitespace-nowrap text-[10px] font-semibold text-rose-600">
   249→              <span className="h-1 w-1 rounded-full bg-rose-500" />Terlambat
   250→            </span>
   251→          )}
   252→        </p>
   253→      )}
   254→      {!o.nextAction && overdue && (
   255→        <p className="mt-2 inline-flex items-center gap-1 rounded bg-rose-50 px-1.5 py-0.5 text-[10px] font-semibold text-rose-600">
   256→          <span className="h-1 w-1 rounded-full bg-rose-500" />Terlambat
   257→        </p>
   258→      )}
   259→
   260→      <div className="mt-2 flex items-center gap-1.5 border-t border-slate-100 pt-2">
   261→        <UserAvatar name={o.ownerName} color={o.ownerColor} size={20} />
   262→        <span className="truncate text-[10px] font-medium text-slate-500">{o.ownerName}</span>
   263→        <span className="ml-auto truncate text-[10px] text-slate-400">{timeAgo(o.lastInteractionAt)}</span>
   264→        <span title={o.companyCountry} className="text-[11px] leading-none">{flag(o.companyCountry)}</span>
   265→      </div>
   266→    </div>
   267→  )
   268→}

   271→export default function PipelineView({ focusOpportunityId, onConsumeFocus }: {
   272→  focusOpportunityId: string | null
   273→  onConsumeFocus: () => void
   274→}) {
   275→  const { toast } = useToast()
   276→  const { brands, user } = useCrmStore()
   277→
   278→  const [opps, setOpps] = useState<OpportunityDTO[]>([])
   279→  const [loading, setLoading] = useState(true)
   280→  const [fetchError, setFetchError] = useState<string | null>(null)
   281→
   282→  /* filters */
   283→  const [search, setSearch] = useState('')
   284→  const [brandFilter, setBrandFilter] = useState('all')
   285→  const [ownerFilter, setOwnerFilter] = useState('all')
   286→  const [tempFilter, setTempFilter] = useState('all')
   287→  const [priorityFilter, setPriorityFilter] = useState('all')
   288→  const [view, setView] = useState<'kanban' | 'tabel'>('kanban')
   289→
   290→  /* owners (MARKETING only) */
   291→  const [marketingUsers, setMarketingUsers] = useState<UserDTO[]>([])
   292→
   293→  /* kanban drag & drop */
   294→  const [dragOver, setDragOver] = useState<string | null>(null)
   295→  const [busyId, setBusyId] = useState<string | null>(null)
   296→
   297→  /* detail drawer */
   298→  const [detailId, setDetailId] = useState<string | null>(null)
   299→  const [detailOpen, setDetailOpen] = useState(false)
   300→
   301→  /* table */
   302→  const [selected, setSelected] = useState<Set<string>>(new Set())
   303→  const [sort, setSort] = useState<SortState>(null)
   304→  const [bulkOwner, setBulkOwner] = useState('')
   305→  const [bulkBusy, setBulkBusy] = useState(false)
   306→
   307→  /* stage dialogs */
   308→  const [lostTarget, setLostTarget] = useState<OpportunityDTO | null>(null)
   309→  const [lostReason, setLostReason] = useState('')
   310→  const [lostNotes, setLostNotes] = useState('')
   311→  const [lostCompetitor, setLostCompetitor] = useState('')
   312→  const [lostReactivation, setLostReactivation] = useState('')
   313→  const [lostFollowUp, setLostFollowUp] = useState('')
   314→  const [wonTarget, setWonTarget] = useState<OpportunityDTO | null>(null)
   315→  const [wonOffer, setWonOffer] = useState('')
   316→
   317→  /* new opportunity dialog */
   318→  const [newOpen, setNewOpen] = useState(false)
   319→  const [creating, setCreating] = useState(false)
   320→  const [form, setForm] = useState<NewOppForm>(EMPTY_FORM)
   321→  const [companies, setCompanies] = useState<CompanyDTO[]>([])
   322→  const [companiesLoading, setCompaniesLoading] = useState(false)
   323→  const [contacts, setContacts] = useState<ContactDTO[]>([])
   324→  const [contactsLoading, setContactsLoading] = useState(false)
   325→
   326→  /* ---------- data ---------- */
   327→  const load = useCallback(async () => {
   328→    setLoading(true)
   329→    try {
   330→      const data = await crmApi.opportunities()
   331→      setOpps(data)
   332→      setFetchError(null)
   333→    } catch (e) {
   334→      setFetchError(e instanceof Error ? e.message : 'Gagal memuat pipeline')
   335→    } finally {
   336→      setLoading(false)
   337→    }
   338→  }, [])
   339→
   340→  useEffect(() => { void load() }, [load])
   341→
   342→  useEffect(() => {
   343→    let alive = true
   344→    crmApi.users()
   345→      .then((us) => { if (alive) setMarketingUsers(us.filter((u) => u.isActive && u.role === 'MARKETING')) })
   346→      .catch(() => { /* biarkan kosong */ })
   347→    return () => { alive = false }
   348→  }, [])
   349→
   350→  /* focus dari view lain (mis. related opportunity di drawer) */
   351→  useEffect(() => {
   352→    if (focusOpportunityId) {
   353→      setDetailId(focusOpportunityId)
   354→      setDetailOpen(true)
   355→      onConsumeFocus()
   356→    }
   357→  }, [focusOpportunityId, onConsumeFocus])
   358→
   359→  /* ---------- client-side filter ---------- */
   360→  const filtered = useMemo(() => {
   361→    const q = search.trim().toLowerCase()
   362→    return opps.filter((o) => {
   363→      if (brandFilter !== 'all' && o.sourceBrandId !== brandFilter) return false
   364→      if (ownerFilter !== 'all' && o.ownerId !== ownerFilter) return false
   365→      if (tempFilter !== 'all' && o.temperature !== tempFilter) return false
   366→      if (priorityFilter !== 'all' && o.priority !== priorityFilter) return false
   367→      if (q) {
   368→        const hay = `${o.code} ${o.title} ${o.companyName} ${o.contactName}`.toLowerCase()
   369→        if (!hay.includes(q)) return false
   370→      }
   371→      return true
   372→    })
   373→  }, [opps, search, brandFilter, ownerFilter, tempFilter, priorityFilter])
   374→
   375→  /* ---------- lead scoring (rule-based, Phase 4) ---------- */
   376→  const scoreMap = useMemo(() => {
   377→    const m = new Map<string, ReturnType<typeof computeLeadScore>>()
   378→    for (const o of filtered) m.set(o.id, computeLeadScore(o))
   379→    return m
   380→  }, [filtered])
   381→
   382→  const tableRows = useMemo(() => {
   383→    const rows = [...filtered]
   384→    if (sort) {
   385→      rows.sort((a, b) => {
   386→        let d = 0
   387→        if (sort.key === 'value') {
   388→          d = a.estimatedValue - b.estimatedValue
   389→        } else if (sort.key === 'score') {
   390→          d = (scoreMap.get(a.id)?.score ?? 0) - (scoreMap.get(b.id)?.score ?? 0)
   391→        } else {
   392→          const ta = a.expectedCloseDate ? new Date(a.expectedCloseDate).getTime() : 0
   393→          const tb = b.expectedCloseDate ? new Date(b.expectedCloseDate).getTime() : 0
   394→          d = ta - tb
   395→        }
   396→        return sort.dir === 'asc' ? d : -d
   397→      })
   398→    }
   399→    return rows
   400→  }, [filtered, sort, scoreMap])
   401→
   402→  /* ---------- stage moves ---------- */
   403→  const applyStageChange = useCallback(async (id: string, stage: Stage, extra?: Record<string, unknown>, successTitle?: string) => {
   404→    setBusyId(id)
   405→    try {
   406→      await crmApi.changeStage(id, { stage, ...extra })
   407→      toast({
   408→        title: successTitle ?? 'Stage diperbarui',
   409→        description: `Opportunity dipindah ke ${stageMeta(stage).label}`,
   410→      })
   411→      await load()
   412→    } catch (e) {
   413→      toast({ title: 'Gagal memindahkan opportunity', description: e instanceof Error ? e.message : 'Terjadi kesalahan', variant: 'destructive' })
   414→    } finally {
   415→      setBusyId(null)
   416→    }
   417→  }, [load, toast])
   418→
   419→  const handleMove = useCallback((o: OpportunityDTO, target: Stage) => {
   420→    if (target === o.stage) return
   421→    if (target === 'LOST') {
   422→      setLostReason(''); setLostNotes(''); setLostCompetitor(''); setLostReactivation(''); setLostFollowUp('')
   423→      setLostTarget(o)
   424→      return
   425→    }
   426→    if (target === 'WON') { setWonOffer(''); setWonTarget(o); return }
   427→    void applyStageChange(o.id, target)
   428→  }, [applyStageChange])
   429→
   430→  const submitLost = () => {
   431→    if (!lostTarget || !lostReason) return
   432→    const id = lostTarget.id
   433→    setLostTarget(null)
   434→    void applyStageChange(id, 'LOST', {
   435→      lostReason,
   436→      ...(lostNotes.trim() ? { lostNotes: lostNotes.trim() } : {}),
   437→      ...(lostCompetitor.trim() ? { competitorName: lostCompetitor.trim() } : {}),
   438→      ...(lostReactivation ? { reactivation: lostReactivation } : {}),
   439→      ...(lostFollowUp ? { followUpDate: lostFollowUp } : {}),
   440→    }, 'Opportunity ditandai Lost')
   441→  }
   442→
   443→  const confirmWon = () => {
   444→    if (!wonTarget) return
   445→    const id = wonTarget.id
   446→    const offer = Number(wonOffer)
   447→    setWonTarget(null)
   448→    void applyStageChange(id, 'WON', offer > 0 ? { lastOfferValue: offer } : {}, '🎉 Deal Won! Project produksi dibuat otomatis')
   449→  }
   450→
   451→  /* ---------- table helpers ---------- */
   452→  const openDetail = useCallback((id: string) => { setDetailId(id); setDetailOpen(true) }, [])
   453→
   454→  const toggleSort = (key: 'value' | 'close' | 'score') => {
   455→    setSort((s) => {
   456→      if (s?.key !== key) return { key, dir: 'desc' }
   457→      if (s.dir === 'desc') return { key, dir: 'asc' }
   458→      return null
   459→    })
   460→  }
   461→
   462→  const toggleRow = (id: string, checked: boolean) => {
   463→    setSelected((prev) => { const n = new Set(prev); if (checked) n.add(id); else n.delete(id); return n })
   464→  }
