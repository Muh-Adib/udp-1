    43→
    44→const rpFormatter = new Intl.NumberFormat("id-ID", { style: "currency", currency: "IDR", maximumFractionDigits: 0 });
    45→
    46→function formatRp(n: number): string {
    47→  return rpFormatter.format(Number.isFinite(n) ? Math.round(n) : 0);
    48→}
    49→
    50→function formatDate(iso?: string | null): string {
    51→  if (!iso) return "—";
    52→  return new Date(iso).toLocaleDateString("id-ID", { day: "numeric", month: "short", year: "numeric" });
    53→}

    83→

    88→  const [formLeadId, setFormLeadId] = useState("");
    89→  const [formTitle, setFormTitle] = useState("");
    90→  const [formItems, setFormItems] = useState<DraftItem[]>([emptyDraftItem()]);

    92→  const [formPpn, setFormPpn] = useState("11");
    93→  const [formNotes, setFormNotes] = useState("");
    94→
    95→  // dialog catat pembayaran
    96→  const [payTarget, setPayTarget] = useState<InvoiceDTO | null>(null);
    97→  const [payAmount, setPayAmount] = useState("0");
    98→  const [payMethod, setPayMethod] = useState("TRANSFER");
    99→  const [payNote, setPayNote] = useState("");
   100→
   101→  const role = user.role;
   102→  const canCreate = role === "OWNER" || role === "MANAGER" || role === "MARKETER";
   103→  const canSend = canCreate;

   109→  const load = useCallback(async () => {
   110→    setLoading(true);

   113→      const [s, q, i] = await Promise.all([api.financeStats(), api.quotations(), api.invoices()]);

   115→      setQuotations(q.quotations);
   116→      setInvoices(i.invoices);

   118→      const msg = (e as Error).message || "Gagal memuat data keuangan";
   119→      setError(msg);
   120→      if (stats || quotations || invoices) toast.error(msg);
   121→    } finally {
   122→      setLoading(false);
   123→    }
   124→    // eslint-disable-next-line react-hooks/exhaustive-deps
   125→  }, []);
   126→
   127→  useEffect(() => {
   128→    void load();

   133→    try {
   134→      const [s, q, i] = await Promise.all([api.financeStats(), api.quotations(), api.invoices()]);
   135→      setStats(s.stats);
   136→      setQuotations(q.quotations);
   137→      setInvoices(i.invoices);

   139→      toast.error((e as Error).message || "Gagal memuat ulang data");

   144→

   146→    setBusy(`send-${q.id}`);

   148→      await api.updateQuotationStatus(q.id, "send");
   149→      toast.success(`Penawaran ${q.number} dikirim ke klien`);
   150→      await refreshData();
   151→    } catch (e) {
   152→      toast.error((e as Error).message);
   153→    } finally {
   154→      setBusy(null);
   155→    }
   156→  }
   157→
   158→  async function approveQuotation(q: QuotationDTO) {
   159→    setBusy(`approve-${q.id}`);
   160→    try {
   161→      const r = await api.updateQuotationStatus(q.id, "approve");

   163→        toast.success(`Penawaran disetujui — Proyek ${r.projectCode} & invoice ${r.invoiceNumber} otomatis dibuat`);

   165→        toast.success(`Penawaran ${q.number} disetujui`);
   166→      }
   167→      await refreshData();
   168→    } catch (e) {
   169→      toast.error((e as Error).message);
   170→    } finally {
   171→      setBusy(null);
   172→    }
   173→  }
   174→
   175→  function openReject(q: QuotationDTO) {
   176→    setRejectTarget(q);
   177→    setRejectNote("");
   178→  }
   179→
   180→  async function submitReject() {
   181→    if (!rejectTarget) return;
   182→    setBusy("reject");
   183→    try {
   184→      await api.updateQuotationStatus(rejectTarget.id, "reject", rejectNote.trim() || undefined);
   185→      toast.success(`Penawaran ${rejectTarget.number} ditolak`);
   186→      setRejectTarget(null);
   187→      await refreshData();
   188→    } catch (e) {
   189→      toast.error((e as Error).message);
   190→    } finally {
   191→      setBusy(null);
   192→    }
   193→  }

   195→  /* ---------- dialog buat penawaran ---------- */
   196→
   197→  function openCreate() {
   198→    setCreateOpen(true);
   199→    // fetch sekali saat dialog pertama kali dibuka
   200→    if (leadsCache === null) {
   201→      setLeadsLoading(true);
   202→      api
   203→        .leads()
   204→        .then(({ leads }) => setLeadsCache(leads))
   205→        .catch((e) => toast.error((e as Error).message))
   206→        .finally(() => setLeadsLoading(false));
   207→    }

   210→  function resetCreateForm() {
   211→    setFormLeadId("");
   212→    setFormTitle("");
   213→    setFormItems([emptyDraftItem()]);
   214→    setFormDiscount("0");
   215→    setFormPpn("11");
   216→    setFormNotes("");
   217→  }
   218→
   219→  const formSubtotal = formItems.reduce((s, it) => s + (Number(it.qty) || 0) * (Number(it.price) || 0), 0);
   220→  const formDiscountVal = formSubtotal * ((Number(formDiscount) || 0) / 100);
   221→  const formAfterDiscount = formSubtotal - formDiscountVal;
   222→  const formPpnVal = formAfterDiscount * ((Number(formPpn) || 0) / 100);
   223→  const formGrandTotal = formAfterDiscount + formPpnVal;
   224→
   225→  async function submitCreate() {
   226→    if (!formLeadId) {
   227→      toast.error("Pilih lead terlebih dahulu");
   228→      return;
   229→    }
   230→    const title = formTitle.trim();
   231→    if (!title) {
   232→      toast.error("Judul penawaran wajib diisi");
   233→      return;
   234→    }
   235→    const items = formItems.map((it) => ({
   236→      desc: it.desc.trim(),
   237→      qty: Math.max(0, Math.floor(Number(it.qty) || 0)),
   238→      price: Math.max(0, Math.round(Number(it.price) || 0)),
   239→    }));
   240→    if (items.some((it) => !it.desc)) {
   241→      toast.error("Deskripsi setiap item wajib diisi");
   242→      return;
   243→    }
   244→    if (items.some((it) => it.qty < 1)) {
   245→      toast.error("Qty setiap item minimal 1");
   246→      return;
   247→    }
   248→    if (formSubtotal <= 0) {
   249→      toast.error("Total penawaran harus lebih dari 0");
   250→      return;
   251→    }
   252→    setBusy("create");
   253→    try {
   254→      const { quotation } = await api.createQuotation({
   255→        leadId: formLeadId,
   256→        title,
   257→        items,
   258→        discountPct: Number(formDiscount) || 0,
   259→        ppnPct: Number(formPpn) || 0,
   260→        notes: formNotes.trim() || undefined,
   261→      });
   262→      toast.success(`Penawaran ${quotation.number} dibuat (draf)`);
   263→      setCreateOpen(false);
   264→      resetCreateForm();
   265→      await refreshData();
   266→    } catch (e) {
   267→      toast.error((e as Error).message);
   268→    } finally {
   269→      setBusy(null);
   270→    }
   271→  }
   272→
   273→  /* ---------- dialog catat pembayaran ---------- */
   274→
   275→  function openPay(inv: InvoiceDTO) {
   276→    setPayTarget(inv);
   277→    setPayAmount(String(Math.max(0, inv.grandTotal - inv.paidAmount)));
   278→    setPayMethod("TRANSFER");
   279→    setPayNote("");
   280→  }
   281→
   282→  async function submitPay() {
   283→    if (!payTarget) return;
   284→    const amount = Math.round(Number(payAmount) || 0);
   285→    const remaining = payTarget.grandTotal - payTarget.paidAmount;
   286→    if (amount <= 0) {
   287→      toast.error("Nominal pembayaran tidak valid");
   288→      return;
   289→    }
   290→    if (amount > remaining) {
   291→      toast.error(`Nominal melebihi sisa tagihan (${formatRp(remaining)})`);
   292→      return;
   293→    }
   294→    setBusy(`pay-${payTarget.id}`);
   295→    try {
   296→      await api.addPayment(payTarget.id, { amount, method: payMethod, note: payNote.trim() || undefined });
   297→      toast.success(`Pembayaran ${formatRp(amount)} untuk ${payTarget.number} tercatat`);
   298→      setPayTarget(null);
   299→      await refreshData();
   300→    } catch (e) {
   301→      toast.error((e as Error).message);
   302→    } finally {
   303→      setBusy(null);
   304→    }
   305→  }
   306→
   307→  /* ---------- render: error ---------- */
   308→
   309→  if (error && (!stats || !quotations || !invoices)) {

   314→            <TriangleAlert className="size-6" />
   315→          </span>
   316→          <div>
   317→            <p className="font-semibold">Gagal memuat data keuangan</p>
   318→            <p className="text-sm text-muted-foreground">{error}</p>
   319→          </div>
   320→          <Button onClick={() => void load()}>
   321→            <RefreshCw className="size-4" /> Coba Lagi
   322→          </Button>
   323→        </CardContent>
   324→      </Card>
   325→    );
   326→  }
   327→
   328→  /* ---------- render: skeleton ---------- */
   329→
   330→  if (loading || !stats || !quotations || !invoices) {
   331→    return (
   332→      <div className="space-y-5">
   333→        <div className="animate-pulse space-y-2">
   334→          <div className="h-6 w-40 rounded bg-muted" />
   335→          <div className="h-3 w-72 rounded bg-muted" />
   336→        </div>
   337→        <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
   338→          {[0, 1, 2, 3].map((i) => (
   339→            <Card key={i}>
   340→              <CardContent className="animate-pulse space-y-3 pt-5">
   341→                <div className="h-3 w-24 rounded bg-muted" />
   342→                <div className="h-7 w-36 rounded bg-muted" />
   343→                <div className="h-3 w-28 rounded bg-muted" />
   344→              </CardContent>
   345→            </Card>
   346→          ))}
   347→        </div>
   348→        <div className="grid gap-4 lg:grid-cols-5">
   349→          <Card className="lg:col-span-3">
   350→            <CardContent className="animate-pulse space-y-3 pt-5">
   351→              <div className="h-4 w-44 rounded bg-muted" />
   352→              <div className="h-40 rounded-xl bg-muted" />
   353→            </CardContent>
   354→          </Card>
   355→          <Card className="lg:col-span-2">
   356→            <CardContent className="animate-pulse space-y-3 pt-5">
   357→              <div className="h-4 w-40 rounded bg-muted" />
   358→              {[0, 1, 2].map((i) => (
   359→                <div key={i} className="h-8 rounded-lg bg-muted" />
   360→              ))}
   361→            </CardContent>
   362→          </Card>
   363→        </div>
   364→        <Card>
   365→          <CardContent className="animate-pulse space-y-3 pt-5">
   366→            <div className="h-9 w-56 rounded-lg bg-muted" />
   367→            {[0, 1, 2, 3, 4].map((i) => (
   368→              <div key={i} className="h-10 rounded-lg bg-muted" />
   369→            ))}
   370→          </CardContent>
   371→        </Card>
   372→      </div>
   373→    );
   374→  }
   375→
   376→  /* ---------- data siap ---------- */
   377→
   378→  const maxCash = Math.max(1, ...stats.monthly.map((m) => Math.max(m.revenue, m.invoiced)));
   379→  const barHeight = (v: number) => {
   380→    const px = Math.round((v / maxCash) * 160);
   381→    return `${Math.max(v > 0 ? 6 : 3, px)}px`;
   382→  };
   383→  const maxBrandRevenue = Math.max(1, ...stats.byBrand.map((b) => b.revenue));
   384→
   385→  const kpis = [
   386→    {
   387→      key: "revenue",
   388→      label: "Pendapatan",
   389→      value: formatRp(stats.revenuePaid),
   390→      sub: "Total pembayaran masuk",
   391→      icon: Banknote,
   392→      iconClass: "bg-emerald-100 text-emerald-700",
   393→      alert: false,
   394→    },
   395→    {
   396→      key: "outstanding",
   397→      label: "Outstanding",
   398→      value: formatRp(stats.outstanding),
   399→      sub: `Belum dibayar dari ${stats.invoiceCount} invoice`,
   400→      icon: Wallet,
   401→      iconClass: "bg-amber-100 text-amber-700",
   402→      alert: false,
   403→    },
   404→    {
   405→      key: "overdue",
   406→      label: "Invoice Jatuh Tempo",
   407→      value: String(stats.overdueCount),
   408→      sub: stats.overdueCount > 0 ? "Segera tindak lanjuti penagihan" : "Semua tagihan terkendali",
   409→      icon: AlarmClock,
   410→      iconClass: stats.overdueCount > 0 ? "bg-rose-100 text-rose-600" : "bg-stone-200 text-stone-600",
   411→      alert: stats.overdueCount > 0,
   412→    },
   413→    {
   414→      key: "approval",
   415→      label: "Tingkat Persetujuan",
   416→      value: `${Math.round(stats.quotationApprovedPct)}%`,
   417→      sub: `Dari ${stats.quotationCount} penawaran`,
   418→      icon: CheckCircle2,
   419→      iconClass: "bg-emerald-100 text-emerald-700",
   420→      alert: false,
   421→    },
   422→  ];
   423→
   424→  return (
   425→    <div className="space-y-5">
   426→      {/* Header */}
   427→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   428→        <div>
   429→          <h2 className="text-xl font-bold tracking-tight">Keuangan</h2>
   430→          <p className="text-sm text-muted-foreground">
   431→            Penawaran, invoice, dan arus kas — gabungan seluruh brand PT. Unicam Digital Pictvres.
   432→          </p>
   433→        </div>
   434→        <Button variant="outline" size="sm" onClick={() => void load()} disabled={loading}>
   435→          <RefreshCw className={cn("size-4", loading && "animate-spin")} /> Muat Ulang
   436→        </Button>
   437→      </div>
   438→
   439→      {/* KPI row */}
   440→      <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
   441→        {kpis.map((k) => {
   442→          const Icon = k.icon;
   443→          return (

   445→              <CardContent className="flex items-start justify-between gap-3 pt-5">
   446→                <div className="min-w-0 space-y-1">

   449→                  <p className="text-xs text-muted-foreground">{k.sub}</p>
   450→                  {k.alert && (

   453→                </div>
   454→                <span className={cn("flex size-10 shrink-0 items-center justify-center rounded-xl", k.iconClass)}>
   455→                  <Icon className="size-5" />
   456→                </span>
   457→              </CardContent>
   458→            </Card>
   459→          );
   460→        })}
   461→      </div>

   463→      {/* Charts */}
   464→      <div className="grid gap-4 lg:grid-cols-5">
   465→        {/* Cashflow 6 bulan — grouped bar CSS murni */}
   466→        <Card className="lg:col-span-3">
   467→          <CardHeader className="flex-row items-start justify-between space-y-0">
   468→            <div className="space-y-1">
   469→              <CardTitle className="text-base">Arus Kas 6 Bulan</CardTitle>
   470→              <CardDescription>Pembayaran diterima vs nilai invoice yang ditagihkan</CardDescription>
   471→            </div>
   472→            <div className="flex shrink-0 items-center gap-3 text-xs text-muted-foreground">
   473→              <span className="flex items-center gap-1.5">
   474→                <span className="size-2.5 rounded-sm bg-emerald-500" /> Diterima
   475→              </span>
   476→              <span className="flex items-center gap-1.5">
   477→                <span className="size-2.5 rounded-sm bg-amber-400" /> Ditagihkan
   478→              </span>

   482→            <div className="flex items-end gap-2 sm:gap-4">
   483→              {stats.monthly.map((m) => (
   484→                <div key={m.month} className="flex flex-1 flex-col items-center gap-2">
   485→                  <div className="flex h-40 w-full items-end justify-center gap-1 border-b border-border/70">

   492→                      title={`${m.label} — Ditagihkan: ${formatRp(m.invoiced)}`}
