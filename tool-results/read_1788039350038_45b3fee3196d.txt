    55→    55→/** Jam respons terbaca: < 48 jam → "X jam", ≥ 48 jam → "Y hari" (1 desimal, koma Indonesia) */
    56→    56→function formatHours(hours: number | null | undefined): string {
    57→    57→  if (hours === null || hours === undefined || Number.isNaN(hours)) return '—'
    58→    58→  if (hours >= 48) return `${fmtNum(hours / 24)} hari`
    59→    59→  return `${fmtNum(hours)} jam`
    60→    60→}

    62→    62→/** Export CSV analitik percakapan — SATU file multi-seksi via shared downloadCsv (; separator + BOM). */
    63→    63→function exportConversationsCsv(data: ConversationAnalyticsDTO) {
    64→    64→  const { kpi, perBrand, perMarketer, channelMix, weekly } = data
    65→    65→  const totalInteractions = perBrand.reduce((acc, b) => acc + b.interactions, 0)
    66→    66→  const rows: (string | number | null | undefined)[][] = [
    67→    67→    // Seksi 1 — KPI
    68→    68→    ['Indikator', 'Nilai'],
    69→    69→    ['Respons Pertama (rata-rata)', formatHours(kpi.avgFirstResponseHours)],
    70→    70→    ['Respons Pertama (median)', formatHours(kpi.medianFirstResponseHours)],
    71→    71→    ['Kepatuhan SLA', kpi.slaCompliancePct !== null ? `${kpi.slaCompliancePct}%` : '—'],
    72→    72→    ['Percakapan Aktif', kpi.totalConversations],
    73→    73→    ['Menunggu Balasan', kpi.unansweredNow],
    74→    74→    ['Total Interaksi', totalInteractions],
    75→    75→    [''],
    76→    76→    // Seksi 2 — per brand
    77→    77→    ['brand', 'avg_first_response_hours', 'sla_pct', 'interactions'],
    78→    78→    ...perBrand.map((b) => [b.brandName, b.firstResponseHours ?? '', b.slaPct ?? '', b.interactions]),
    79→    79→    [''],
    80→    80→    // Seksi 3 — campuran kanal
    81→    81→    ['channel', 'count'],
    82→    82→    ...channelMix.map((c) => [channelMeta(c.channel).label, c.count]),
    83→    83→    [''],
    84→    84→    // Seksi 4 — aktivitas mingguan
    85→    85→    ['label', 'inbound', 'outbound'],
    86→    86→    ...weekly.map((w) => [w.label, w.inbound, w.outbound]),
    87→    87→    [''],
    88→    88→    // Seksi 5 — per marketer

    90→    90→    ...perMarketer.map((m) => [m.userName, m.replies, m.avgResponseHours ?? '']),

    92→    92→  downloadCsv(`laporan-percakapan-${new Date().toISOString().slice(0, 10)}.csv`, ['Laporan Analitik Percakapan', 'Grupa Kreasi CRM'], rows)

   576→   576→function dueInfo(task: TaskDTO): { label: string; overdue: boolean } {
   577→   577→  if (!task.dueDate) return { label: 'Tanpa tenggat', overdue: false }
   578→   578→  const d = daysUntil(task.dueDate)
   579→   579→  if (d === null) return { label: formatDate(task.dueDate), overdue: false }
   580→   580→  if (d < 0) return { label: `${formatDate(task.dueDate)} · terlambat ${Math.abs(d)} hari`, overdue: true }
   581→   581→  if (d === 0) return { label: `${formatDate(task.dueDate)} · hari ini`, overdue: false }
   582→   582→  return { label: `${formatDate(task.dueDate)} · ${d} hari lagi`, overdue: false }
   583→   583→}
